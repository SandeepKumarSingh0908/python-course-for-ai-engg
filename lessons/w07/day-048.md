## Section 1: KMeans Clustering

KMeans groups data into **K clusters** by iteratively assigning points to the nearest centroid and updating centroids.

### How KMeans Works

```
1. Choose K random points as initial centroids
2. Assign each data point to the nearest centroid
3. Recalculate centroids as the mean of assigned points
4. Repeat steps 2-3 until centroids stop moving (convergence)
```

### Basic KMeans

```python
import numpy as np
from sklearn.cluster import KMeans
from sklearn.datasets import make_blobs

# Generate synthetic clustered data
X, y_true = make_blobs(
    n_samples=300, centers=4, cluster_std=0.60, random_state=42
)

# Fit KMeans
kmeans = KMeans(n_clusters=4, random_state=42, n_init=10)
kmeans.fit(X)

# Results
labels = kmeans.labels_                # cluster assignment for each point
centroids = kmeans.cluster_centers_    # coordinates of cluster centers
inertia = kmeans.inertia_             # sum of squared distances to centroids

print(f"Cluster labels: {np.unique(labels)}")
print(f"Points per cluster: {np.bincount(labels)}")
print(f"Centroids shape: {centroids.shape}")
print(f"Inertia: {inertia:.2f}")

# Predict cluster for new data
new_points = np.array([[0, 0], [5, 5]])
predictions = kmeans.predict(new_points)
print(f"New point clusters: {predictions}")
```

### Choosing K: The Elbow Method

The optimal K is where adding more clusters stops significantly reducing inertia.

```python
from sklearn.cluster import KMeans
from sklearn.datasets import make_blobs

X, _ = make_blobs(n_samples=300, centers=4, cluster_std=0.60, random_state=42)

inertias = []
K_range = range(1, 11)

for k in K_range:
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    km.fit(X)
    inertias.append(km.inertia_)

print(f"{'K':<5} {'Inertia':>12} {'Decrease':>12}")
print("-" * 31)
for i, (k, inertia) in enumerate(zip(K_range, inertias)):
    decrease = inertias[i-1] - inertia if i > 0 else 0
    print(f"{k:<5} {inertia:>12.2f} {decrease:>12.2f}")
```

You will see a sharp drop in inertia from K=1 to K=4, then much smaller drops. The "elbow" at K=4 indicates the optimal number of clusters.

### Silhouette Score

Measures how similar a point is to its own cluster compared to other clusters. Ranges from -1 (wrong cluster) to +1 (well-clustered).

```python
from sklearn.metrics import silhouette_score, silhouette_samples
import numpy as np

X, _ = make_blobs(n_samples=300, centers=4, cluster_std=0.60, random_state=42)

for k in range(2, 8):
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = km.fit_predict(X)
    score = silhouette_score(X, labels)
    print(f"K={k}: Silhouette = {score:.4f}")

# Per-sample silhouette scores
km = KMeans(n_clusters=4, random_state=42, n_init=10)
labels = km.fit_predict(X)
sample_scores = silhouette_samples(X, labels)

for cluster in range(4):
    cluster_scores = sample_scores[labels == cluster]
    print(f"Cluster {cluster}: mean silhouette = {cluster_scores.mean():.4f}, "
          f"size = {len(cluster_scores)}")
```

## Section 2: DBSCAN -- Density-Based Clustering

DBSCAN finds clusters of **arbitrary shape** based on point density. Unlike KMeans, it does not require specifying K and can identify **outliers** (noise points).

### Key Parameters

- **eps**: Maximum distance between two points to be considered neighbors
- **min_samples**: Minimum points required to form a dense region (core point)

```python
from sklearn.cluster import DBSCAN
from sklearn.datasets import make_moons
import numpy as np

# Generate crescent-shaped clusters (KMeans fails on these)
X, y_true = make_moons(n_samples=300, noise=0.05, random_state=42)

# DBSCAN
db = DBSCAN(eps=0.2, min_samples=5)
labels = db.fit_predict(X)

n_clusters = len(set(labels)) - (1 if -1 in labels else 0)
n_noise = (labels == -1).sum()

print(f"Clusters found: {n_clusters}")
print(f"Noise points:   {n_noise}")
print(f"Labels:         {np.unique(labels)}")  # -1 = noise

# Compare with KMeans on the same data
from sklearn.cluster import KMeans
km_labels = KMeans(n_clusters=2, random_state=42, n_init=10).fit_predict(X)

from sklearn.metrics import adjusted_rand_score
print(f"\nDBSCAN ARI:  {adjusted_rand_score(y_true, labels):.4f}")
print(f"KMeans ARI:  {adjusted_rand_score(y_true, km_labels):.4f}")
# DBSCAN should score much higher on crescent-shaped data
```

### KMeans vs DBSCAN

| Feature | KMeans | DBSCAN |
|---|---|---|
| Must specify K? | Yes | No (discovers it) |
| Cluster shapes | Spherical | Arbitrary |
| Handles noise? | No (assigns all) | Yes (labels as -1) |
| Scalability | Very fast | Slower on large data |
| Sensitive to | K, initialization | eps, min_samples |

## Section 3: PCA -- Dimensionality Reduction

**Principal Component Analysis** finds the axes (principal components) that capture the most variance in your data. It reduces dimensions while preserving as much information as possible.

### Why Reduce Dimensions?

- **Visualization**: Plot 100-dimensional data in 2D
- **Speed**: Fewer features = faster training
- **Noise reduction**: Remove low-variance noise dimensions
- **Feature engineering**: Create powerful combined features

### Basic PCA

```python
from sklearn.decomposition import PCA
from sklearn.datasets import load_iris
import numpy as np

X, y = load_iris(return_X_y=True)
print(f"Original shape: {X.shape}")  # (150, 4)

# Reduce from 4 dimensions to 2
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X)
print(f"Reduced shape:  {X_pca.shape}")  # (150, 2)

# How much information did we retain?
print(f"\nExplained variance ratio: {pca.explained_variance_ratio_}")
print(f"Total variance retained: {pca.explained_variance_ratio_.sum():.4f}")
# ~97% of the variance is captured by just 2 components!
```

### Choosing the Number of Components

```python
from sklearn.decomposition import PCA
from sklearn.datasets import load_digits
import numpy as np

X, y = load_digits(return_X_y=True)
print(f"Original dimensions: {X.shape[1]}")  # 64

# Fit PCA with all components to see the variance profile
pca_full = PCA()
pca_full.fit(X)

cumulative_variance = np.cumsum(pca_full.explained_variance_ratio_)

print(f"\n{'Components':<12} {'Cumulative Variance':>20}")
print("-" * 34)
for n in [5, 10, 20, 30, 40, 50, 64]:
    print(f"{n:<12} {cumulative_variance[n-1]:>20.4f}")
```

### PCA with Variance Threshold

```python
# Automatically choose components to retain 95% of variance
pca_95 = PCA(n_components=0.95)
X_reduced = pca_95.fit_transform(X)

print(f"Components needed for 95% variance: {pca_95.n_components_}")
print(f"Reduced shape: {X_reduced.shape}")
```

## Section 4: Visualizing Clusters in 2D

Combine PCA + clustering to visualize high-dimensional clusters.

```python
import numpy as np
from sklearn.datasets import load_digits
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score, adjusted_rand_score

# ----------------------------------------------------------
# Step 1: Load high-dimensional data
# ----------------------------------------------------------
digits = load_digits()
X, y_true = digits.data, digits.target
print(f"Original shape: {X.shape}")  # (1797, 64)

# ----------------------------------------------------------
# Step 2: Reduce to 2D with PCA
# ----------------------------------------------------------
pca = PCA(n_components=2, random_state=42)
X_2d = pca.fit_transform(X)
print(f"Reduced shape: {X_2d.shape}")
print(f"Variance retained: {pca.explained_variance_ratio_.sum():.2%}")

# ----------------------------------------------------------
# Step 3: Cluster in original space
# ----------------------------------------------------------
kmeans = KMeans(n_clusters=10, random_state=42, n_init=10)
cluster_labels = kmeans.fit_predict(X)

print(f"\nSilhouette score: {silhouette_score(X, cluster_labels):.4f}")
print(f"Adjusted Rand Index: {adjusted_rand_score(y_true, cluster_labels):.4f}")

# ----------------------------------------------------------
# Step 4: Visualize with matplotlib
# ----------------------------------------------------------
import matplotlib.pyplot as plt

fig, axes = plt.subplots(1, 2, figsize=(16, 6))

# Left: true labels
scatter1 = axes[0].scatter(X_2d[:, 0], X_2d[:, 1], c=y_true, cmap="tab10", s=5, alpha=0.7)
axes[0].set_title("True Labels (PCA 2D)")
axes[0].set_xlabel("PC1")
axes[0].set_ylabel("PC2")
plt.colorbar(scatter1, ax=axes[0])

# Right: KMeans clusters
scatter2 = axes[1].scatter(X_2d[:, 0], X_2d[:, 1], c=cluster_labels, cmap="tab10", s=5, alpha=0.7)
axes[1].set_title("KMeans Clusters (PCA 2D)")
axes[1].set_xlabel("PC1")
axes[1].set_ylabel("PC2")
plt.colorbar(scatter2, ax=axes[1])

plt.tight_layout()
plt.savefig("cluster_visualization.png", dpi=100)
plt.show()
```

### PCA Before Clustering (Practical Pattern)

For very high-dimensional data, applying PCA first can improve clustering speed and quality.

```python
from sklearn.pipeline import Pipeline
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.datasets import load_digits
from sklearn.metrics import silhouette_score

X = load_digits().data

# Compare: clustering with and without PCA
results = {}

for n_components in [None, 50, 30, 10]:
    if n_components is None:
        km = KMeans(n_clusters=10, random_state=42, n_init=10)
        labels = km.fit_predict(X)
        label = f"No PCA ({X.shape[1]}d)"
    else:
        pipe = Pipeline([
            ("pca", PCA(n_components=n_components, random_state=42)),
            ("kmeans", KMeans(n_clusters=10, random_state=42, n_init=10))
        ])
        labels = pipe.fit_predict(X)
        label = f"PCA ({n_components}d)"

    score = silhouette_score(X, labels)
    results[label] = score
    print(f"{label:<20} Silhouette: {score:.4f}")
```

## Section 5: Worked Example -- Customer Segmentation

A realistic example of unsupervised learning for business intelligence.

```python
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score

# ----------------------------------------------------------
# Step 1: Generate synthetic customer data
# ----------------------------------------------------------
np.random.seed(42)
n_customers = 500

df = pd.DataFrame({
    "annual_income": np.concatenate([
        np.random.normal(30000, 5000, 150),   # low income
        np.random.normal(60000, 8000, 200),   # mid income
        np.random.normal(120000, 15000, 150)  # high income
    ]),
    "spending_score": np.concatenate([
        np.random.normal(70, 10, 150),   # high spenders (budget shoppers)
        np.random.normal(50, 15, 200),   # moderate spenders
        np.random.normal(30, 8, 150)     # low spenders (savers)
    ]),
    "age": np.concatenate([
        np.random.normal(25, 5, 150),
        np.random.normal(40, 10, 200),
        np.random.normal(55, 8, 150)
    ]),
    "purchase_frequency": np.concatenate([
        np.random.normal(15, 3, 150),
        np.random.normal(8, 4, 200),
        np.random.normal(4, 2, 150)
    ])
})

print(df.describe())

# ----------------------------------------------------------
# Step 2: Scale the features
# ----------------------------------------------------------
scaler = StandardScaler()
X_scaled = scaler.fit_transform(df)

# ----------------------------------------------------------
# Step 3: Find optimal K using elbow + silhouette
# ----------------------------------------------------------
print(f"\n{'K':<5} {'Inertia':>12} {'Silhouette':>12}")
print("-" * 31)
for k in range(2, 8):
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = km.fit_predict(X_scaled)
    sil = silhouette_score(X_scaled, labels)
    print(f"{k:<5} {km.inertia_:>12.2f} {sil:>12.4f}")

# ----------------------------------------------------------
# Step 4: Fit with optimal K
# ----------------------------------------------------------
optimal_k = 3
km_final = KMeans(n_clusters=optimal_k, random_state=42, n_init=10)
df["segment"] = km_final.fit_predict(X_scaled)

# ----------------------------------------------------------
# Step 5: Profile each segment
# ----------------------------------------------------------
print("\nSegment profiles:")
print(df.groupby("segment").agg(["mean", "count"]).round(1))

# ----------------------------------------------------------
# Step 6: Visualize in 2D
# ----------------------------------------------------------
pca = PCA(n_components=2, random_state=42)
X_2d = pca.fit_transform(X_scaled)

print(f"\nPCA variance retained: {pca.explained_variance_ratio_.sum():.2%}")
print(f"PC1 loadings: {dict(zip(df.columns[:4], pca.components_[0].round(3)))}")
print(f"PC2 loadings: {dict(zip(df.columns[:4], pca.components_[1].round(3)))}")
```

## Common Mistakes

1. **Forgetting to scale before KMeans.** KMeans uses Euclidean distance, so unscaled features with large ranges dominate the clustering.

2. **Assuming KMeans always finds the right clusters.** KMeans assumes spherical clusters and can fail badly on non-convex shapes. Try DBSCAN or increase `n_init` for better initialization.

3. **Using PCA on unscaled data.** PCA finds directions of maximum variance. If one feature has a much larger scale, it will dominate the principal components. Always scale first.

4. **Interpreting PCA components as original features.** Each principal component is a linear combination of ALL original features. Examine `pca.components_` to understand what each PC represents.

5. **Evaluating unsupervised learning with supervised metrics.** If you have true labels, you can use ARI (Adjusted Rand Index). If not, use silhouette score. Do not use accuracy for clustering.

## Solutions

### Solution 1: Elbow Method and Silhouette

```python
import numpy as np
from sklearn.datasets import load_wine
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score, adjusted_rand_score

wine = load_wine()
X = wine.data
y_true = wine.target
X_scaled = StandardScaler().fit_transform(X)

print(f"{'K':<5} {'Inertia':>12} {'Silhouette':>12}")
print("-" * 31)
for k in range(2, 9):
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = km.fit_predict(X_scaled)
    sil = silhouette_score(X_scaled, labels)
    print(f"{k:<5} {km.inertia_:>12.2f} {sil:>12.4f}")

# Fit with the true K=3
km3 = KMeans(n_clusters=3, random_state=42, n_init=10)
labels3 = km3.fit_predict(X_scaled)
ari = adjusted_rand_score(y_true, labels3)
print(f"\nAdjusted Rand Index (K=3 vs true labels): {ari:.4f}")
# Expect ARI around 0.90 -- KMeans recovers the wine classes well.
```

### Solution 2: DBSCAN on Moons

```python
import numpy as np
from sklearn.datasets import make_moons
from sklearn.cluster import DBSCAN, KMeans
from sklearn.metrics import adjusted_rand_score

X, y_true = make_moons(n_samples=500, noise=0.1, random_state=42)

# DBSCAN at the default eps
db = DBSCAN(eps=0.2, min_samples=5)
db_labels = db.fit_predict(X)
print(f"DBSCAN (eps=0.2) ARI: {adjusted_rand_score(y_true, db_labels):.4f}")

# KMeans baseline
km_labels = KMeans(n_clusters=2, random_state=42, n_init=10).fit_predict(X)
print(f"KMeans            ARI: {adjusted_rand_score(y_true, km_labels):.4f}")

# Try different eps values
print("\nEps sweep:")
for eps in [0.1, 0.2, 0.3, 0.5]:
    labels = DBSCAN(eps=eps, min_samples=5).fit_predict(X)
    n_clusters = len(set(labels)) - (1 if -1 in labels else 0)
    n_noise = (labels == -1).sum()
    ari = adjusted_rand_score(y_true, labels)
    print(f"  eps={eps}: clusters={n_clusters}, noise={n_noise}, ARI={ari:.4f}")
# eps=0.2 or 0.3 typically gives the best ARI for this dataset.
```

### Solution 3: PCA + Classification

```python
import time
import numpy as np
from sklearn.datasets import load_digits
from sklearn.model_selection import cross_val_score
from sklearn.ensemble import RandomForestClassifier
from sklearn.decomposition import PCA
from sklearn.pipeline import Pipeline

X, y = load_digits(return_X_y=True)
rf = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)

configs = [
    ("No PCA (64 features)", None),
    ("PCA (30 components)", 30),
    ("PCA (10 components)", 10),
]

print(f"{'Config':<25} {'Mean Acc':>10} {'Time (s)':>10}")
print("-" * 47)
for label, n_components in configs:
    if n_components is None:
        model = rf
    else:
        model = Pipeline([
            ("pca", PCA(n_components=n_components, random_state=42)),
            ("clf", rf)
        ])
    start = time.time()
    scores = cross_val_score(model, X, y, cv=5, scoring="accuracy")
    elapsed = time.time() - start
    print(f"{label:<25} {scores.mean():>10.4f} {elapsed:>10.2f}")

# PCA(30) is usually very close in accuracy to no PCA, but faster.
# PCA(10) reduces accuracy more noticeably -- too much information is lost.
```
