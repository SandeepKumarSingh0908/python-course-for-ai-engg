## Section 1: Lambda Expressions

A `lambda` is an anonymous, single-expression function.

```python
# Named function
def square(x):
    return x * x

# Equivalent lambda
square = lambda x: x * x

print(square(5))  # 25
```

Lambdas are most useful as throwaway functions passed to higher-order functions. If the logic needs more than one expression, write a regular `def`.

```python
# Good: short, clear intent
sorted_words = sorted(["banana", "apple", "cherry"], key=lambda w: len(w))
print(sorted_words)  # ['apple', 'banana', 'cherry']

# Bad: lambda too complex -- use a named function instead
# process = lambda x: x.strip().lower().replace("-", "_") if x else ""
```

## Section 2: map(), filter(), and reduce()

These three functions are the foundation of functional data transformation.

### map() -- Transform Every Element

```python
# Apply a function to every item in an iterable
temperatures_f = [32, 68, 100, 212]
temperatures_c = list(map(lambda f: (f - 32) * 5 / 9, temperatures_f))
print(temperatures_c)  # [0.0, 20.0, 37.78, 100.0]

# map with multiple iterables
names = ["alice", "bob", "charlie"]
scores = [95, 82, 91]
records = list(map(lambda n, s: {"name": n, "score": s}, names, scores))
print(records)
# [{'name': 'alice', 'score': 95}, {'name': 'bob', 'score': 82}, ...]
```

### filter() -- Keep Elements That Pass a Test

```python
# Keep only even numbers
nums = range(10)
evens = list(filter(lambda x: x % 2 == 0, nums))
print(evens)  # [0, 2, 4, 6, 8]

# Filter out empty strings from a dataset
raw_texts = ["hello", "", "world", "  ", "test", ""]
non_empty = list(filter(lambda t: t.strip(), raw_texts))
print(non_empty)  # ['hello', 'world', 'test']
```

### reduce() -- Collapse to a Single Value

`reduce` is in `functools` (not a builtin since Python 3).

```python
from functools import reduce

# Sum of all elements
total = reduce(lambda acc, x: acc + x, [1, 2, 3, 4, 5])
print(total)  # 15

# Flatten a list of lists
nested = [[1, 2], [3, 4], [5, 6]]
flat = reduce(lambda acc, lst: acc + lst, nested, [])
print(flat)  # [1, 2, 3, 4, 5, 6]

# Build a word frequency dict
words = ["the", "cat", "sat", "on", "the", "mat", "the"]
freq = reduce(
    lambda acc, w: {**acc, w: acc.get(w, 0) + 1},
    words,
    {}
)
print(freq)  # {'the': 3, 'cat': 1, 'sat': 1, 'on': 1, 'mat': 1}
```

### map/filter vs List Comprehensions

Python developers often prefer comprehensions for simple cases.

```python
# These are equivalent:
result = list(map(lambda x: x**2, filter(lambda x: x > 0, data)))
result = [x**2 for x in data if x > 0]  # preferred for readability
```

**Guideline:** Use comprehensions for simple transforms and filters. Use `map`/`filter` when passing an existing named function or when composing in a pipeline.

## Section 3: functools -- Power Tools

### functools.partial -- Fix Some Arguments

```python
from functools import partial

def train_model(data, lr=0.001, epochs=10, batch_size=32):
    print(f"Training: lr={lr}, epochs={epochs}, batch_size={batch_size}")
    return "trained"

# Create specialized versions
quick_train = partial(train_model, epochs=3, batch_size=64)
fine_tune = partial(train_model, lr=0.0001, epochs=20)

quick_train("my_data")   # Training: lr=0.001, epochs=3, batch_size=64
fine_tune("my_data")     # Training: lr=0.0001, epochs=20, batch_size=32
```

### functools.lru_cache -- Memoization

Cache the return value of expensive function calls.

```python
from functools import lru_cache
import time

@lru_cache(maxsize=128)
def compute_embedding(text: str) -> list:
    """Simulate an expensive embedding computation."""
    time.sleep(0.1)  # simulate work
    return [hash(text) % 100 / 100.0]  # fake embedding

# First call: slow
start = time.time()
compute_embedding("hello world")
print(f"First call: {time.time() - start:.3f}s")   # ~0.1s

# Second call: instant (cached)
start = time.time()
compute_embedding("hello world")
print(f"Cached call: {time.time() - start:.6f}s")   # ~0.000s

# Cache info
print(compute_embedding.cache_info())
# CacheInfo(hits=1, misses=1, maxsize=128, currsize=1)
```

**Caution:** Only use `lru_cache` on functions with hashable arguments. Lists and dicts are not hashable.

## Section 4: itertools -- Efficient Iteration

`itertools` provides memory-efficient building blocks for iteration. These return iterators, not lists, so they work well with large datasets.

### chain -- Concatenate Iterables

```python
from itertools import chain

train_data = [1, 2, 3]
val_data = [4, 5]
test_data = [6, 7]

all_data = list(chain(train_data, val_data, test_data))
print(all_data)  # [1, 2, 3, 4, 5, 6, 7]

# chain.from_iterable for a list of lists
batches = [[10, 20], [30, 40], [50]]
flat = list(chain.from_iterable(batches))
print(flat)  # [10, 20, 30, 40, 50]
```

### product -- Cartesian Product (Hyperparameter Grid)

```python
from itertools import product

learning_rates = [0.001, 0.0001]
batch_sizes = [16, 32, 64]
optimizers = ["adam", "sgd"]

grid = list(product(learning_rates, batch_sizes, optimizers))
print(f"Total combinations: {len(grid)}")  # 12

for lr, bs, opt in grid:
    print(f"  lr={lr}, batch_size={bs}, optimizer={opt}")
```

### combinations and permutations

```python
from itertools import combinations, permutations

features = ["age", "income", "education", "experience"]

# All pairs of features (order does not matter)
for pair in combinations(features, 2):
    print(pair)
# ('age', 'income'), ('age', 'education'), ... (6 total)

# All orderings of 2 features (order matters)
print(len(list(permutations(features, 2))))  # 12
```

### groupby -- Group Consecutive Elements

```python
from itertools import groupby

# Data MUST be sorted by the grouping key first
records = [
    {"label": "positive", "text": "great"},
    {"label": "positive", "text": "awesome"},
    {"label": "negative", "text": "terrible"},
    {"label": "negative", "text": "awful"},
    {"label": "neutral", "text": "okay"},
]

for label, group in groupby(records, key=lambda r: r["label"]):
    items = list(group)
    print(f"{label}: {len(items)} records")
# positive: 2 records
# negative: 2 records
# neutral: 1 records
```

### islice, takewhile, dropwhile

```python
from itertools import islice, takewhile, dropwhile

# Take first N items from any iterator (like head)
data = range(1_000_000)
first_five = list(islice(data, 5))
print(first_five)  # [0, 1, 2, 3, 4]

# Take while condition is true
losses = [2.5, 2.1, 1.8, 1.2, 1.5, 1.9, 2.0]
improving = list(takewhile(lambda x: x > 1.3, losses))
print(improving)  # [2.5, 2.1, 1.8]

# Drop while condition is true (skip warmup)
after_warmup = list(dropwhile(lambda x: x > 2.0, losses))
print(after_warmup)  # [1.8, 1.2, 1.5, 1.9, 2.0]
```

## Section 5: Functional vs Imperative -- When to Use Each

### Prefer Functional When

- Applying a single transformation to every element
- Chaining multiple independent steps
- Building a hyperparameter grid (product)
- Caching pure function results (lru_cache)
- Working with infinite or very large iterators

### Prefer Imperative When

- Logic requires multiple conditions and early exits
- You need to modify state across iterations
- Readability suffers from deeply nested map/filter/lambda
- Debugging -- stepping through a for-loop is easier than tracing through chained iterators

```python
# Functional: clean and concise
pipeline = map(str.lower, filter(bool, map(str.strip, raw_lines)))

# Imperative: clearer when logic is complex
results = []
for line in raw_lines:
    line = line.strip()
    if not line:
        continue
    if line.startswith("#"):  # skip comments
        continue
    tokens = line.lower().split()
    if len(tokens) < 3:
        continue
    results.append(tokens)
```

## Section 6: Worked Example -- Data Transformation Pipeline

```python
from functools import reduce, partial
from itertools import chain, groupby

# Raw training data: list of (text, label) tuples from multiple sources
source_a = [
    ("  Great product!  ", "positive"),
    ("Terrible experience.", "negative"),
    ("", "positive"),           # empty text -- bad record
    ("It was okay.", "neutral"),
]
source_b = [
    ("Love it!!!", "positive"),
    ("  ", "negative"),          # whitespace only -- bad record
    ("Not worth the price.", "negative"),
]

# Step 1: Combine sources
all_data = list(chain(source_a, source_b))

# Step 2: Clean text
clean = map(lambda pair: (pair[0].strip().lower(), pair[1]), all_data)

# Step 3: Filter out empty texts
valid = filter(lambda pair: pair[0], clean)

# Step 4: Convert to dicts
records = list(map(lambda pair: {"text": pair[0], "label": pair[1]}, valid))
print(f"Valid records: {len(records)}")  # 5

# Step 5: Group by label and count
records_sorted = sorted(records, key=lambda r: r["label"])
for label, group in groupby(records_sorted, key=lambda r: r["label"]):
    count = sum(1 for _ in group)
    print(f"  {label}: {count}")

# Step 6: Build a label-to-index mapping
labels = sorted(set(r["label"] for r in records))
label_to_idx = {label: i for i, label in enumerate(labels)}
print(label_to_idx)  # {'negative': 0, 'neutral': 1, 'positive': 2}
```

## Common Mistakes

| Mistake | Fix |
|---------|-----|
| Using `lambda` for complex logic | Write a named function; lambdas are for simple expressions |
| Forgetting `map()` and `filter()` return iterators | Wrap in `list()` if you need to use the result more than once |
| Using `groupby` on unsorted data | Sort by the grouping key first; `groupby` only groups consecutive elements |
| Overusing `reduce` for things that have builtins | Use `sum()`, `max()`, `min()`, `any()`, `all()` when they fit |
| Nesting too many lambdas | Extract named helper functions for clarity |
| Mutating arguments in a "functional" pipeline | Return new objects instead of modifying inputs |

## Solutions

### Solution 1: Filter Users Under 18

```python
from functools import reduce

raw_users = [
    {"name": "Alice", "age": 28, "purchases": [100, 200, 50], "active": True},
    {"name": "Bob", "age": 17, "purchases": [], "active": True},
    {"name": "Charlie", "age": 35, "purchases": [500, 300, 200, 100], "active": False},
    {"name": "Diana", "age": 22, "purchases": [50], "active": True},
    {"name": "Eve", "age": 45, "purchases": [1000, 2000, 500], "active": True},
    {"name": "Frank", "age": 15, "purchases": [20, 10], "active": True},
    {"name": "Grace", "age": 31, "purchases": [150, 250, 100], "active": True},
]

adults = list(filter(lambda u: u["age"] >= 18, raw_users))
assert len(adults) == 5
```

### Solution 2: Filter Inactive Users

```python
active_adults = list(filter(lambda u: u["active"], adults))
assert len(active_adults) == 4
```

### Solution 3: Add Computed Features

```python
def add_features(user):
    user = user.copy()
    purchases = user["purchases"]
    user["total_spent"] = sum(purchases)
    user["avg_purchase"] = sum(purchases) / len(purchases) if purchases else 0
    user["purchase_count"] = len(purchases)
    return user

enriched = list(map(add_features, active_adults))
```

### Solution 4: Sort by Total Spent Descending

```python
sorted_users = sorted(enriched, key=lambda u: u["total_spent"], reverse=True)
# Eve is the top spender
assert sorted_users[0]["name"] == "Eve"
```

### Solution 5: Aggregate Stats with reduce()

```python
eligible_users = sorted_users  # 4 users

total_revenue = reduce(lambda acc, u: acc + u["total_spent"], eligible_users, 0)
average_age = reduce(lambda acc, u: acc + u["age"], eligible_users, 0) / len(eligible_users)

print(f"Total revenue: {total_revenue}")
print(f"Average age: {average_age:.1f}")
```

### Solution 6: Build Feature Vectors

```python
feature_vectors = list(map(
    lambda u: [u["age"], u["total_spent"], round(u["avg_purchase"], 2), u["purchase_count"]],
    eligible_users,
))

print(feature_vectors)
# Eve (highest spender) should be first
assert feature_vectors[0] == [45, 3500, round(3500 / 3, 2), 3]
print("Pipeline complete!")
```
