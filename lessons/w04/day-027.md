## Section 1: The logging Module

### Why Not print()?

`print()` fails in production:
- No severity levels (is this info or a critical error?)
- No timestamps
- Cannot redirect to files, log servers, or monitoring systems
- Cannot filter by importance
- Hard to turn off without deleting lines

### Logging Levels

| Level | Numeric | When to Use |
|-------|---------|-------------|
| `DEBUG` | 10 | Detailed info for diagnosing problems |
| `INFO` | 20 | Confirmation that things work as expected |
| `WARNING` | 30 | Something unexpected but not breaking (default level) |
| `ERROR` | 40 | A serious problem; some function failed |
| `CRITICAL` | 50 | The program may not be able to continue |

### Basic Setup

```python
import logging

# Quick config (good for scripts)
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S",
)

logger = logging.getLogger(__name__)

logger.debug("This won't show -- level is INFO")
logger.info("Training started")
logger.warning("GPU memory usage at 90%%")
logger.error("Failed to load checkpoint: file not found")
logger.critical("CUDA device not available -- aborting")
```

Output:

```
2024-03-15 10:00:01 [INFO] __main__: Training started
2024-03-15 10:00:01 [WARNING] __main__: GPU memory usage at 90%
2024-03-15 10:00:01 [ERROR] __main__: Failed to load checkpoint: file not found
2024-03-15 10:00:01 [CRITICAL] __main__: CUDA device not available -- aborting
```

### Logging with Variables

```python
# Use %-style formatting (lazy evaluation -- only formats if the message is emitted)
logger.info("Epoch %d/%d - loss: %.4f, acc: %.4f", epoch, total, loss, acc)

# Also acceptable: f-strings (always evaluated, even if level is filtered out)
logger.info(f"Epoch {epoch}/{total} - loss: {loss:.4f}, acc: {acc:.4f}")
```

Prefer `%`-style for hot paths where the message might not be emitted; use f-strings elsewhere for readability.

### Handlers and Formatters

```python
import logging

def setup_logging(log_file: str = "training.log") -> logging.Logger:
    """Configure logging to both console and file."""
    logger = logging.getLogger("ml_pipeline")
    logger.setLevel(logging.DEBUG)  # capture everything; handlers filter

    # Console handler: INFO and above
    console = logging.StreamHandler()
    console.setLevel(logging.INFO)
    console_fmt = logging.Formatter(
        "[%(levelname)s] %(message)s"
    )
    console.setFormatter(console_fmt)

    # File handler: DEBUG and above (more detail)
    file_handler = logging.FileHandler(log_file, encoding="utf-8")
    file_handler.setLevel(logging.DEBUG)
    file_fmt = logging.Formatter(
        "%(asctime)s [%(levelname)s] %(name)s:%(lineno)d - %(message)s"
    )
    file_handler.setFormatter(file_fmt)

    logger.addHandler(console)
    logger.addHandler(file_handler)

    return logger

logger = setup_logging()
logger.debug("Detailed debug info -- only in file")
logger.info("Visible on console and in file")
```

logger = logging.getLogger(__name__)

## Section 2: Debugging with pdb and breakpoint()

### breakpoint() -- The Modern Way

Since Python 3.7, `breakpoint()` drops you into the debugger at that line.

```python
def compute_metrics(predictions, labels):
    precision = sum(p == l for p, l in zip(predictions, labels)) / len(predictions)
    breakpoint()  # execution pauses here
    recall = precision  # placeholder -- inspect and fix
    f1 = 2 * precision * recall / (precision + recall)
    return {"precision": precision, "recall": recall, "f1": f1}
```

### Essential pdb Commands

| Command | Short | Action |
|---------|-------|--------|
| `help` | `h` | List commands |
| `next` | `n` | Execute next line (step over) |
| `step` | `s` | Step into a function call |
| `continue` | `c` | Continue to next breakpoint |
| `print expr` | `p expr` | Print the value of an expression |
| `pretty-print` | `pp expr` | Pretty-print (useful for dicts/lists) |
| `list` | `l` | Show source code around current line |
| `where` | `w` | Show the call stack |
| `up` / `down` | `u` / `d` | Navigate the call stack |
| `quit` | `q` | Exit the debugger |

### Practical Debugging Session

```python
def process_batch(batch):
    results = []
    for item in batch:
        # Suspect: something goes wrong here
        score = item["score"] / item["weight"]
        results.append(score)
    return results

data = [
    {"score": 10, "weight": 2},
    {"score": 20, "weight": 0},   # will cause ZeroDivisionError
    {"score": 30, "weight": 5},
]

# Add breakpoint to investigate
def process_batch_debug(batch):
    results = []
    for i, item in enumerate(batch):
        if item["weight"] == 0:
            breakpoint()  # pause when we hit the problem case
        score = item["score"] / item["weight"]
        results.append(score)
    return results
```

## Section 3: Profiling with cProfile and timeit

### timeit -- Micro-Benchmarks

Use `timeit` to compare small code snippets.

```python
import timeit

# Compare list comprehension vs map
setup = "data = list(range(10000))"

time_comp = timeit.timeit("[x**2 for x in data]", setup=setup, number=1000)
time_map = timeit.timeit("list(map(lambda x: x**2, data))", setup=setup, number=1000)

print(f"Comprehension: {time_comp:.4f}s")
print(f"Map + lambda:  {time_map:.4f}s")
```

### timeit from the Command Line

```bash
python -m timeit -s "data = list(range(10000))" "[x**2 for x in data]"
```

### cProfile -- Full Program Profiling

```python
import cProfile

def slow_function():
    total = 0
    for i in range(1_000_000):
        total += i ** 0.5
    return total

def main():
    results = []
    for _ in range(5):
        results.append(slow_function())
    return sum(results)

# Profile the entire program
cProfile.run("main()")
```

Output shows:

```
         8 function calls in 0.523 seconds

   Ordered by: standard name

   ncalls  tottime  percall  cumtime  percall filename:lineno(function)
        5    0.521    0.104    0.521    0.104 script.py:3(slow_function)
        1    0.002    0.002    0.523    0.523 script.py:9(main)
        ...
```

Key columns:
- `ncalls`: how many times the function was called
- `tottime`: total time spent in the function (excluding sub-calls)
- `cumtime`: cumulative time (including sub-calls)

### Profiling to a File and Analyzing

```python
import cProfile
import pstats

# Save profile data
cProfile.run("main()", "profile_output.prof")

# Analyze
stats = pstats.Stats("profile_output.prof")
stats.sort_stats("cumulative")
stats.print_stats(10)  # top 10 functions by cumulative time
```

### Context Manager for Timing Code Blocks

```python
import time

class Timer:
    """Simple context-manager timer for profiling code blocks."""
    def __init__(self, label: str = ""):
        self.label = label
        self.elapsed: float = 0.0

    def __enter__(self):
        self.start = time.perf_counter()
        return self

    def __exit__(self, *args):
        self.elapsed = time.perf_counter() - self.start
        if self.label:
            print(f"[{self.label}] {self.elapsed:.4f}s")

# Usage
with Timer("data loading"):
    data = list(range(1_000_000))

with Timer("processing"):
    result = [x ** 2 for x in data]
```

## Section 4: The traceback Module

When you catch an exception but want to log the full traceback without crashing:

```python
import traceback
import logging

logger = logging.getLogger(__name__)

def risky_operation():
    return 1 / 0

try:
    risky_operation()
except ZeroDivisionError:
    # Get traceback as a string for logging
    tb_str = traceback.format_exc()
    logger.error("Operation failed:\n%s", tb_str)

    # Or print directly
    traceback.print_exc()
```

### Extracting Traceback Programmatically

```python
import traceback
import sys

try:
    risky_operation()
except ZeroDivisionError:
    exc_type, exc_value, exc_tb = sys.exc_info()
    frames = traceback.extract_tb(exc_tb)
    for frame in frames:
        print(f"  File {frame.filename}, line {frame.lineno}, in {frame.name}")
        print(f"    {frame.line}")
```

## Section 5: Worked Example -- Instrumented Training Loop

```python
import logging
import time
import cProfile
from dataclasses import dataclass, field

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    datefmt="%H:%M:%S",
)
logger = logging.getLogger("trainer")

@dataclass
class TrainingMetrics:
    epoch: int
    loss: float
    accuracy: float
    duration_seconds: float

def simulate_training_step(batch_size: int) -> tuple[float, float]:
    """Simulate a training step returning (loss, accuracy)."""
    import random
    time.sleep(0.01)  # simulate compute
    loss = random.uniform(0.1, 2.0)
    accuracy = random.uniform(0.5, 0.99)
    return loss, accuracy

def train(epochs: int = 5, batches_per_epoch: int = 10, batch_size: int = 32):
    """Training loop with logging and timing."""
    logger.info("Starting training: %d epochs, %d batches/epoch", epochs, batches_per_epoch)
    all_metrics: list[TrainingMetrics] = []

    for epoch in range(1, epochs + 1):
        epoch_start = time.perf_counter()
        epoch_loss = 0.0
        epoch_acc = 0.0

        for batch in range(batches_per_epoch):
            try:
                loss, acc = simulate_training_step(batch_size)
                epoch_loss += loss
                epoch_acc += acc
                logger.debug(
                    "Epoch %d, Batch %d: loss=%.4f, acc=%.4f",
                    epoch, batch, loss, acc,
                )
            except Exception:
                logger.exception("Error in epoch %d, batch %d", epoch, batch)
                continue

        avg_loss = epoch_loss / batches_per_epoch
        avg_acc = epoch_acc / batches_per_epoch
        duration = time.perf_counter() - epoch_start

        metrics = TrainingMetrics(epoch, avg_loss, avg_acc, duration)
        all_metrics.append(metrics)

        logger.info(
            "Epoch %d/%d - loss: %.4f, acc: %.4f (%.2fs)",
            epoch, epochs, avg_loss, avg_acc, duration,
        )

        # Warn if loss is suspiciously high
        if avg_loss > 1.5:
            logger.warning("High loss at epoch %d: %.4f", epoch, avg_loss)

    logger.info("Training complete. Final loss: %.4f", all_metrics[-1].loss)
    return all_metrics

if __name__ == "__main__":
    # Normal run
    metrics = train(epochs=3, batches_per_epoch=5)

    # Profiled run
    print("\n--- Profile ---")
    cProfile.run("train(epochs=2, batches_per_epoch=5)")
```

**5. A final summary log**
Total time, records loaded, records transformed, records saved.

## Common Mistakes

| Mistake | Fix |
|---------|-----|
| Using `print()` for logging in production code | Use the `logging` module |
| Calling `logging.basicConfig()` multiple times | It only takes effect the first time; use `dictConfig` or manual handler setup |
| Setting root logger to DEBUG in production | Use INFO for console, DEBUG for file only |
| Leaving `breakpoint()` calls in committed code | Use `PYTHONBREAKPOINT=0` in production, and add a linter rule |
| Profiling with `time.time()` instead of `time.perf_counter()` | `perf_counter()` has higher resolution and is not affected by clock adjustments |
| Not using `logger.exception()` in except blocks | It automatically includes the traceback; saves manual `traceback.format_exc()` |

## Solutions

### Solution: Instrumented Data Pipeline

```python
import csv
import json
import logging
import time
from pathlib import Path


# ── Logging setup ──────────────────────────────────────────────────────────────

def setup_logging(log_file: str = "pipeline.log") -> logging.Logger:
    """Configure logging to console (INFO) and file (DEBUG)."""
    logger = logging.getLogger("pipeline")
    logger.setLevel(logging.DEBUG)

    console = logging.StreamHandler()
    console.setLevel(logging.INFO)
    console.setFormatter(logging.Formatter("[%(levelname)s] %(message)s"))

    fh = logging.FileHandler(log_file, encoding="utf-8")
    fh.setLevel(logging.DEBUG)
    fh.setFormatter(
        logging.Formatter("%(asctime)s [%(levelname)s] %(name)s:%(lineno)d - %(message)s")
    )

    logger.addHandler(console)
    logger.addHandler(fh)
    return logger


logger = setup_logging()


# ── Timer context manager ──────────────────────────────────────────────────────

class Timer:
    def __init__(self, label: str = ""):
        self.label = label
        self.elapsed: float = 0.0

    def __enter__(self):
        self.start = time.perf_counter()
        return self

    def __exit__(self, *args):
        self.elapsed = time.perf_counter() - self.start
        if self.label:
            logger.debug("[%s] %.4fs", self.label, self.elapsed)


# ── Pipeline functions ─────────────────────────────────────────────────────────

def create_sample_data(path: str, num_rows: int = 100) -> None:
    """Create a sample CSV for testing."""
    import random
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["id", "text", "score"])
        for i in range(num_rows):
            if random.random() < 0.05:
                writer.writerow([i, "", random.randint(1, 5)])
            elif random.random() < 0.05:
                writer.writerow([i, f"Sample text {i}", "not_a_number"])
            else:
                writer.writerow([i, f"Sample text {i}", random.randint(1, 5)])


def load_data(path: str) -> list[dict]:
    """Load CSV data with error handling and logging."""
    file_path = Path(path)

    try:
        file_size = file_path.stat().st_size
    except FileNotFoundError:
        logger.error("File not found: %s", path)
        raise

    logger.info("Loading %s (%.1f KB)", path, file_size / 1024)

    records = []
    skipped = 0

    with Timer("load_data"):
        try:
            with open(file_path, encoding="utf-8") as f:
                reader = csv.DictReader(f)
                for i, row in enumerate(reader, start=2):
                    try:
                        text = row["text"].strip()
                        if not text:
                            raise ValueError("Empty text field")
                        score = int(row["score"])
                        records.append({"id": int(row["id"]), "text": text, "score": score})
                        logger.debug("Row %d loaded: id=%s", i, row["id"])
                    except (KeyError, ValueError) as e:
                        logger.warning("Row %d skipped: %s", i, e)
                        skipped += 1
        except UnicodeDecodeError as e:
            logger.error("Encoding error in %s: %s", path, e)
            raise

    logger.info(
        "Load complete: %d valid rows, %d skipped",
        len(records), skipped,
    )
    return records


def transform_data(records: list[dict]) -> list[dict]:
    """Transform records: normalize text, categorize scores."""
    logger.info("Transforming %d records...", len(records))
    transformed = []
    errors = 0

    with Timer("transform_data"):
        for record in records:
            try:
                record = record.copy()
                record["text"] = record["text"].lower().strip()
                record["category"] = "high" if record["score"] >= 4 else "low"
                transformed.append(record)
            except Exception as e:
                logger.error("Transform error for record %s: %s", record.get("id"), e)
                errors += 1

    logger.info("Transform complete: %d records, %d errors", len(transformed), errors)
    return transformed


def save_results(records: list[dict], path: str) -> None:
    """Save results to JSON Lines format."""
    logger.info("Saving %d records to %s...", len(records), path)

    with Timer("save_results"):
        try:
            with open(path, "w", encoding="utf-8") as f:
                for record in records:
                    f.write(json.dumps(record) + "\n")
        except OSError as e:
            logger.error("Failed to write %s: %s", path, e)
            raise

    logger.info("Save complete: %s", path)


def run_pipeline():
    """Run the full pipeline with timing and observability."""
    logger.info("Pipeline starting...")
    pipeline_start = time.perf_counter()

    try:
        create_sample_data("sample_data.csv", num_rows=1000)

        with Timer("full pipeline"):
            data = load_data("sample_data.csv")
            transformed = transform_data(data)
            save_results(transformed, "results.jsonl")

        total_time = time.perf_counter() - pipeline_start
        logger.info(
            "Pipeline complete in %.2fs | loaded=%d, transformed=%d, saved=%d",
            total_time, len(data), len(transformed), len(transformed),
        )

    except Exception:
        logger.exception("Pipeline failed")
        raise


if __name__ == "__main__":
    run_pipeline()
```
