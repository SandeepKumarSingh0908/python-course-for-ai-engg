## Section 1: Closures -- The Foundation

A closure is a function that remembers variables from the enclosing scope even after that
scope has finished executing.

```python
def make_multiplier(factor: float):
    """Returns a function that multiplies its input by factor."""
    def multiply(x: float) -> float:
        return x * factor  # 'factor' is captured from the enclosing scope
    return multiply

double = make_multiplier(2)
triple = make_multiplier(3)

print(double(5))   # 10
print(triple(5))   # 15

# The closure remembers 'factor' even though make_multiplier has returned
print(double.__closure__[0].cell_contents)  # 2
```

### Why closures matter for decorators

A decorator is just a function that:
1. Takes a function as input
2. Returns a new function (the wrapper) that adds behavior
3. The wrapper is a closure that remembers the original function

## Section 2: Basic Function Decorators

### The pattern

```python
import time

def timer(func):
    """Decorator that prints how long a function takes to run."""
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)  # call the original function
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.4f}s")
        return result
    return wrapper
```

### Using `@` syntax

```python
@timer
def train_model(epochs: int) -> list[float]:
    """Simulate training by sleeping."""
    losses = []
    for i in range(epochs):
        time.sleep(0.01)  # simulate work
        losses.append(1.0 / (i + 1))
    return losses

losses = train_model(5)
# train_model took 0.0523s
```

The `@timer` syntax is exactly equivalent to:

```python
def train_model(epochs: int) -> list[float]:
    ...

train_model = timer(train_model)
```

### The problem: lost metadata

```python
print(train_model.__name__)    # "wrapper"  -- not "train_model"!
print(train_model.__doc__)     # None       -- docstring lost!
```

### The fix: `functools.wraps`

```python
import functools

def timer(func):
    @functools.wraps(func)  # copies __name__, __doc__, etc. from func to wrapper
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.4f}s")
        return result
    return wrapper

@timer
def train_model(epochs: int) -> list[float]:
    """Simulate training by sleeping."""
    losses = []
    for i in range(epochs):
        time.sleep(0.01)
        losses.append(1.0 / (i + 1))
    return losses

print(train_model.__name__)   # "train_model"  -- preserved!
print(train_model.__doc__)    # "Simulate training by sleeping."
```

**Rule:** Always use `@functools.wraps(func)` in your decorators.

## Section 3: Decorators with Arguments

Sometimes you want to configure the decorator itself:

```python
def retry(max_attempts: int = 3, delay: float = 1.0):
    """Decorator factory: retries a function on exception."""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    print(f"Attempt {attempt}/{max_attempts} failed: {e}")
                    if attempt < max_attempts:
                        time.sleep(delay)
            raise last_exception
        return wrapper
    return decorator
```

Notice the three levels of nesting:
1. `retry(...)` -- called with config args, returns `decorator`
2. `decorator(func)` -- receives the function, returns `wrapper`
3. `wrapper(*args, **kwargs)` -- runs every time the decorated function is called

```python
import random

@retry(max_attempts=3, delay=0.1)
def call_llm_api(prompt: str) -> str:
    """Simulate a flaky API call."""
    if random.random() < 0.7:
        raise ConnectionError("API timeout")
    return f"Response to: {prompt}"

# Will retry up to 3 times
try:
    result = call_llm_api("Explain decorators")
    print(result)
except ConnectionError:
    print("All attempts failed")
```

## Section 4: Practical Decorators for AI/ML

### Caching with `@lru_cache`

```python
from functools import lru_cache
import time

@lru_cache(maxsize=128)
def compute_embedding(text: str) -> list[float]:
    """Simulate an expensive embedding computation."""
    time.sleep(0.5)  # simulate API call
    # In real code: return openai.Embedding.create(input=text)
    return [hash(text) % 100 / 100.0 for _ in range(3)]

# First call: slow (0.5s)
start = time.perf_counter()
emb1 = compute_embedding("hello world")
print(f"First call: {time.perf_counter() - start:.3f}s")

# Second call: instant (cached)
start = time.perf_counter()
emb2 = compute_embedding("hello world")
print(f"Cached call: {time.perf_counter() - start:.6f}s")

print(emb1 == emb2)  # True

# Cache info
print(compute_embedding.cache_info())
# CacheInfo(hits=1, misses=1, maxsize=128, currsize=1)
```

### Input validation decorator

```python
def validate_types(**expected_types):
    """Decorator that validates argument types at runtime."""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            for param_name, expected_type in expected_types.items():
                if param_name in bound.arguments:
                    value = bound.arguments[param_name]
                    if not isinstance(value, expected_type):
                        raise TypeError(
                            f"{param_name} must be {expected_type.__name__}, "
                            f"got {type(value).__name__}"
                        )
            return func(*args, **kwargs)
        return wrapper
    return decorator

@validate_types(epochs=int, learning_rate=float)
def train(epochs: int, learning_rate: float) -> str:
    return f"Training for {epochs} epochs at lr={learning_rate}"

print(train(10, 0.01))      # Works
# train("ten", 0.01)        # TypeError: epochs must be int, got str
```

### Logging decorator

```python
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def log_calls(func):
    """Log every call with arguments and return value."""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        args_repr = [repr(a) for a in args]
        kwargs_repr = [f"{k}={v!r}" for k, v in kwargs.items()]
        signature = ", ".join(args_repr + kwargs_repr)
        logger.info(f"Calling {func.__name__}({signature})")
        result = func(*args, **kwargs)
        logger.info(f"{func.__name__} returned {result!r}")
        return result
    return wrapper

@log_calls
def predict(features: list[float]) -> float:
    return sum(features) / len(features)

predict([1.0, 2.0, 3.0])
# INFO: Calling predict([1.0, 2.0, 3.0])
# INFO: predict returned 2.0
```

## Section 5: Built-in Decorators Recap

| Decorator       | Purpose                                 | Defined in       |
|-----------------|-----------------------------------------|------------------|
| `@property`     | Make a method act like an attribute      | builtins         |
| `@classmethod`  | Method receives `cls` instead of `self`  | builtins         |
| `@staticmethod` | Method receives neither `self` nor `cls` | builtins         |
| `@lru_cache`    | Memoize function results                 | `functools`      |
| `@total_ordering`| Fill in comparison methods              | `functools`      |
| `@abstractmethod`| Mark method as abstract                 | `abc`            |
| `@dataclass`    | Auto-generate `__init__`, `__repr__`, etc| `dataclasses`    |

### Stacking decorators

Decorators apply bottom-up:

```python
@timer
@log_calls
def process(data):
    return sorted(data)

# Equivalent to: process = timer(log_calls(process))
# Execution order: timer's wrapper runs first, which calls log_calls' wrapper,
# which calls the original process.
```

## Section 6: Worked Example -- A Complete Caching Layer

```python
import functools
import time
import json

def disk_cache(cache_dir: str = "/tmp/cache"):
    """Cache function results to disk as JSON files."""
    import os
    os.makedirs(cache_dir, exist_ok=True)

    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            # Create a cache key from function name + arguments
            key_data = {"func": func.__name__, "args": args, "kwargs": kwargs}
            key = str(hash(json.dumps(key_data, sort_keys=True, default=str)))
            cache_path = os.path.join(cache_dir, f"{key}.json")

            # Check cache
            if os.path.exists(cache_path):
                with open(cache_path) as f:
                    print(f"Cache HIT for {func.__name__}")
                    return json.load(f)

            # Compute and store
            print(f"Cache MISS for {func.__name__}")
            result = func(*args, **kwargs)
            with open(cache_path, "w") as f:
                json.dump(result, f)
            return result
        return wrapper
    return decorator


@disk_cache(cache_dir="/tmp/ml_cache")
def expensive_feature_extraction(text: str) -> dict:
    """Simulate expensive NLP feature extraction."""
    time.sleep(0.1)
    words = text.lower().split()
    return {
        "word_count": len(words),
        "unique_words": len(set(words)),
        "avg_word_length": sum(len(w) for w in words) / len(words),
    }


features = expensive_feature_extraction("The quick brown fox jumps over the lazy dog")
print(features)
# Cache MISS (first call)

features = expensive_feature_extraction("The quick brown fox jumps over the lazy dog")
print(features)
# Cache HIT (second call)
```

## Common Mistakes

1. **Forgetting `@functools.wraps(func)`** -- the decorated function loses its name,
   docstring, and type hints. This breaks introspection tools and documentation generators.

2. **Missing the extra nesting level** for decorators with arguments. `@retry(3)` needs
   three levels: factory -> decorator -> wrapper. `@timer` (no args) needs only two.

3. **Not returning the result** from the wrapper -- `wrapper` must `return func(*args,
   **kwargs)`, not just call it. Otherwise the decorated function always returns `None`.

4. **Using `@lru_cache` on methods** -- `self` is part of the cache key, so the cache
   is per-instance and never shared. Use `@lru_cache` on module-level functions or
   `@functools.cached_property` for methods.

5. **Decorating with mutable default arguments** -- if your decorator uses a list or dict
   as a default, all decorated functions share that mutable object. Use `None` and create
   inside the function.
