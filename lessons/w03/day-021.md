### Solution 1: Experiment class

```python
from datetime import datetime

class Experiment:
    experiment_count = 0

    def __init__(self, name: str):
        self.name = name
        self.created_at = datetime.now()
        self.runs: list[dict] = []
        Experiment.experiment_count += 1

    def add_run(self, params: dict, score: float) -> None:
        self.runs.append({"params": params, "score": score})

    @property
    def best_run(self) -> dict | None:
        if not self.runs:
            return None
        return max(self.runs, key=lambda r: r["score"])

        return cls(config["name"])

    def __repr__(self) -> str:
        return f"Experiment('{self.name}', runs={len(self.runs)})"

    def __len__(self) -> int:
        return len(self.runs)


exp = Experiment("lr_sweep")
```

### Solution 2: RobustScaler and LogTransformer

```python
from abc import ABC, abstractmethod
import math

class BaseTransformer(ABC):
    @abstractmethod
    def fit(self, data: list[float]) -> "BaseTransformer": ...

    @abstractmethod
    def transform(self, data: list[float]) -> list[float]: ...

    def fit_transform(self, data: list[float]) -> list[float]:
        return self.fit(data).transform(data)


class RobustScaler(BaseTransformer):
    def __init__(self):
        self._median: float = 0.0
        self._iqr: float = 1.0

    def fit(self, data: list[float]) -> "RobustScaler":
        sorted_data = sorted(data)
        n = len(sorted_data)
        # Median
        mid = n // 2
        self._median = (sorted_data[mid - 1] + sorted_data[mid]) / 2 if n % 2 == 0 else sorted_data[mid]
        # Q1 and Q3 (exclusive median method)
        lower_half = sorted_data[:mid]
        upper_half = sorted_data[mid + (0 if n % 2 == 0 else 1):]
        def median_of(lst):
            m = len(lst) // 2
            return (lst[m - 1] + lst[m]) / 2 if len(lst) % 2 == 0 else lst[m]
        q1 = median_of(lower_half)
        q3 = median_of(upper_half)
        self._iqr = q3 - q1 if q3 != q1 else 1.0
        return self

    def transform(self, data: list[float]) -> list[float]:
        return [(x - self._median) / self._iqr for x in data]


class LogTransformer(BaseTransformer):
    def fit(self, data: list[float]) -> "LogTransformer":
        return self   # stateless

    def transform(self, data: list[float]) -> list[float]:
        return [math.log1p(x) for x in data]


data = [1, 2, 3, 4, 5, 100]
scaler = RobustScaler()
scaled = scaler.fit_transform(data)
print(f"Robust scaled: {[round(v, 2) for v in scaled]}")

log_t = LogTransformer()
logged = log_t.fit_transform([0, 1, 10, 100])
print(f"Log transformed: {[round(v, 4) for v in logged]}")
# [0.0, 0.6931, 2.3979, 4.6151]

try:
    BaseTransformer()
except TypeError as e:
    print(f"ABC works: {e}")

print("Exercise 2: PASSED")
```

### Solution 3: Prediction class

```python
class Prediction:
    def __init__(self, label: str, confidence: float):
        self.label = label
        self.confidence = confidence

    def __repr__(self) -> str:
        return f"Prediction('{self.label}', confidence={self.confidence})"

    def __str__(self) -> str:
        return f"{self.label} ({self.confidence * 100:.1f}%)"

        if not isinstance(other, Prediction):
            return NotImplemented
        return self.label == other.label and self.confidence == other.confidence
```

### Solution 4: Decorators

```python
import functools
import time

# 4a: count_calls
def count_calls(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        wrapper.call_count += 1
        return func(*args, **kwargs)
    wrapper.call_count = 0
    return wrapper


@count_calls
def process(x):
    return x * 2

process(1); process(2); process(3)
assert process.call_count == 3


# 4b: cache_results
def cache_results(func):
    @functools.wraps(func)
    def wrapper(*args):
        if args not in wrapper.cache:
            wrapper.cache[args] = func(*args)
        return wrapper.cache[args]

    wrapper.cache = {}
    wrapper.cache_clear = lambda: wrapper.cache.clear()
    return wrapper


@cache_results
def slow_square(x):
    time.sleep(0.01)
    return x ** 2

assert slow_square(4) == 16
assert slow_square(4) == 16   # cached
assert 4 in slow_square.cache
slow_square.cache_clear()
assert len(slow_square.cache) == 0


# 4c: validate_positive
def validate_positive(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        for arg in list(args) + list(kwargs.values()):
            if isinstance(arg, (int, float)) and arg < 0:
                raise ValueError(f"Negative argument: {arg}")
        return func(*args, **kwargs)
    return wrapper


@validate_positive
def compute_loss(predicted: float, actual: float) -> float:
    return abs(predicted - actual)

assert compute_loss(3.0, 2.0) == 1.0
try:
    compute_loss(-1.0, 2.0)
    assert False, "Should have raised"
except ValueError:
    pass

print("Exercise 4: PASSED")
```

### Solution 5: Context Managers

```python
from contextlib import contextmanager
import sys
import io

# 5a: ResourceTracker
class ResourceTracker:
    def __init__(self):
        self.count = 0
        self._start = 0

    def __enter__(self):
        self._start = self.count
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        allocated = self.count - self._start
        print(f"Resources allocated: {allocated}")
        return False


with ResourceTracker() as tracker:
    tracker.count += 10
    tracker.count += 5
# Prints: Resources allocated: 15


# 5b: capture_output
@contextmanager
def capture_output():
    buffer = io.StringIO()
    old_stdout = sys.stdout
    sys.stdout = buffer
    lines: list[str] = []
    try:
        yield lines
    finally:
        sys.stdout = old_stdout
        lines.extend(buffer.getvalue().splitlines())


with capture_output() as output:
    print("hello")
    print("world")
assert output == ["hello", "world"]

print("Exercise 5: PASSED")
```

### Solution 6: Modules & Packages (Conceptual Answers)

**Question 1 — importing DataLoader into train.py**

Using an absolute import (recommended):

```python
from project.data.loader import DataLoader
```

This works when the project is installed (`pip install -e .`) or when the `src/` directory is on `sys.path`.

**Question 2 — does `from your_package import utils` work?**

Yes, `__all__` only restricts `from your_package import *` (star imports). An explicit name import like `from your_package import utils` bypasses `__all__` and succeeds as long as `utils` exists as a module or is defined in `__init__.py`.

**Question 3 — 147 packages from pip freeze**

The problem is that `pip freeze` lists every installed package, including transitive dependencies you never directly depend on. The fix: maintain a hand-written `requirements.txt` (or `pyproject.toml`) with only your direct dependencies, then pin to a lockfile separately for reproducibility.

**Question 4 — relative import fails with `python script.py`**

`python script.py` runs the file as a top-level script (`__name__ == "__main__"`), so Python does not treat it as part of a package. Relative imports (`.utils`) require a package context. `python -m package.script` tells the interpreter to run `script` as part of `package`, establishing the package context so relative imports resolve correctly.
