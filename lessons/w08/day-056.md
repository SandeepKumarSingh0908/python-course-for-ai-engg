## Common Mistakes

1. **Starting with the UI instead of the pipeline** -- Build the RAG pipeline first and test it in a script. Add FastAPI once the pipeline works.
2. **Not testing retrieval before generation** -- If retrieval returns irrelevant chunks, no prompt will save you. Verify retrieval quality first.
3. **Skipping error handling** -- API keys expire, rate limits hit, models return unexpected output. Handle every external call.
4. **Over-engineering the first version** -- Build the simplest possible version first (fixed chunking, top-K retrieval, basic prompt). Then iterate.
5. **Not logging anything** -- Without logs, you cannot debug production issues. Log predictions from day one.

### Solution for Exercise 1: LLM API call with error handling

```python
import os
from openai import OpenAI
from anthropic import Anthropic
import openai as openai_lib
import anthropic as anthropic_lib


def call_llm(provider: str, prompt: str) -> dict:
    """
    Call either OpenAI or Anthropic based on the provider parameter.
    Handles RateLimitError and returns a dict with content and tokens_used.
    """
    if provider == "openai":
        client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
        try:
            response = client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[{"role": "user", "content": prompt}],
                max_tokens=512,
            )
            return {
                "content": response.choices[0].message.content,
                "tokens_used": response.usage.total_tokens,
            }
        except openai_lib.RateLimitError as e:
            print(f"OpenAI rate limited: {e}")
            raise

    elif provider == "anthropic":
        client = Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))
        try:
            message = client.messages.create(
                model="claude-haiku-4-20250514",
                max_tokens=512,
                messages=[{"role": "user", "content": prompt}],
            )
            return {
                "content": message.content[0].text,
                "tokens_used": message.usage.input_tokens + message.usage.output_tokens,
            }
        except anthropic_lib.RateLimitError as e:
            print(f"Anthropic rate limited: {e}")
            raise

    else:
        raise ValueError(f"Unknown provider: {provider}")
```

### Solution for Exercise 3: Semantic search with sentence-transformers and FAISS

```python
import numpy as np
import faiss
from sentence_transformers import SentenceTransformer


def find_similar(query: str, sentences: list[str], top_k: int = 2) -> list[dict]:
    """Find the top_k most similar sentences to the query using FAISS."""
    model = SentenceTransformer("all-MiniLM-L6-v2")

    # Encode and normalize all sentences
    embeddings = model.encode(sentences).astype("float32")
    faiss.normalize_L2(embeddings)

    # Build FAISS index
    dimension = embeddings.shape[1]
    index = faiss.IndexFlatIP(dimension)
    index.add(embeddings)

    # Encode and normalize the query
    query_emb = model.encode([query]).astype("float32")
    faiss.normalize_L2(query_emb)

    # Search
    distances, indices = index.search(query_emb, k=top_k)

    results = []
    for idx, score in zip(indices[0], distances[0]):
        results.append({"text": sentences[idx], "score": float(score)})
    return results


sentences = [
    "Python uses try/except blocks for exception handling.",
    "Lists and tuples are sequence types in Python.",
    "The raise statement triggers an exception manually.",
    "Decorators modify function behavior without changing code.",
    "Logging helps debug issues in production systems.",
]

results = find_similar("How do I handle errors?", sentences, top_k=2)
for r in results:
    print(f"[{r['score']:.4f}] {r['text']}")
```

### Solution for Exercise 4: Sentence-boundary chunking with overlap

```python
import re


def chunk_by_sentence_boundary(
    text: str,
    max_chars: int = 300,
    overlap_sentences: int = 1,
) -> list[str]:
    """
    Split text into chunks of at most max_chars characters,
    breaking only at sentence boundaries (periods), with 1-sentence overlap.
    """
    sentences = re.split(r"(?<=\.)\s+", text.strip())
    sentences = [s.strip() for s in sentences if s.strip()]

    chunks = []
    start = 0

    while start < len(sentences):
        current_chunk = []
        current_len = 0

        i = start
        while i < len(sentences):
            sentence = sentences[i]
            candidate_len = current_len + len(sentence) + (1 if current_chunk else 0)
            if candidate_len <= max_chars or not current_chunk:
                current_chunk.append(sentence)
                current_len = candidate_len
                i += 1
            else:
                break

        chunks.append(" ".join(current_chunk))
        # Move start forward by (number of sentences added - overlap)
        advance = max(1, len(current_chunk) - overlap_sentences)
        start += advance

    return chunks
```

### Solution for Exercise 5: FastAPI /classify endpoint

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

app = FastAPI()


class ClassifyRequest(BaseModel):
    text: str = Field(..., min_length=1, max_length=10_000, description="Text to classify")
    labels: list[str] = Field(..., min_length=2, description="Candidate labels (at least two)")


class ClassifyResponse(BaseModel):
    label: str = Field(..., description="The predicted label")
    confidence: float = Field(..., ge=0.0, le=1.0, description="Confidence score between 0 and 1")


@app.post("/classify", response_model=ClassifyResponse)
async def classify(request: ClassifyRequest):
    """Classify text into one of the provided labels."""
    import os, json
    from openai import AsyncOpenAI

    client = AsyncOpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    labels_str = ", ".join(request.labels)
    prompt = (
        f"Classify the following text into exactly one of these labels: {labels_str}.\n"
        f'Respond with JSON only: {{"label": "<chosen>", "confidence": <0.0-1.0>}}\n\n'
        f"Text: {request.text}"
    )
    try:
        resp = await client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": "You are a text classifier. Respond with valid JSON only."},
                {"role": "user", "content": prompt},
            ],
            response_format={"type": "json_object"},
            max_tokens=64,
            temperature=0.0,
        )
        data = json.loads(resp.choices[0].message.content)
        if data.get("label") not in request.labels:
            raise ValueError(f"Model returned unexpected label: {data.get('label')}")
        return ClassifyResponse(label=data["label"], confidence=float(data["confidence"]))
    except Exception as e:
        raise HTTPException(status_code=502, detail=f"Classification failed: {e}")
```

### Solution for Exercise 6: DriftAlert class

```python
class DriftAlert:
    """
    Detects if a single observed value is an outlier relative to a
    reference distribution described by its mean and standard deviation.
    """

    def __init__(self, reference_mean: float, reference_std: float, z_threshold: float = 3.0):
        """
        Args:
            reference_mean: Mean of the reference (training) distribution.
            reference_std: Standard deviation of the reference distribution.
            z_threshold: Number of standard deviations beyond which a value is flagged.
        """
        self.reference_mean = reference_mean
        self.reference_std = reference_std
        self.z_threshold = z_threshold

    def check(self, value: float) -> bool:
        """
        Return True if the value is more than z_threshold standard deviations
        from the reference mean, False otherwise.
        """
        if self.reference_std == 0:
            return value != self.reference_mean
        z_score = abs(value - self.reference_mean) / self.reference_std
        return z_score > self.z_threshold
```
