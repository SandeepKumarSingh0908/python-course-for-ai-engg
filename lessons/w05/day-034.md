## Section 1: GroupBy -- Split, Apply, Combine

The GroupBy pattern: split the data into groups, apply a function to each group, and combine the results.

```python
import pandas as pd
import numpy as np

sales = pd.DataFrame({
    "region": ["East", "West", "East", "West", "East", "West", "East", "West"],
    "product": ["A", "A", "B", "B", "A", "B", "B", "A"],
    "revenue": [100, 150, 200, 120, 180, 90, 160, 130],
    "quantity": [10, 15, 20, 12, 18, 9, 16, 13],
})
```

### Basic groupby

```python
# Group by region and compute mean revenue
print(sales.groupby("region")["revenue"].mean())
# East    160.0
# West    122.5

# Group by multiple columns
print(sales.groupby(["region", "product"])["revenue"].sum())
# region  product
# East    A          280
#         B          360
# West    A          280
#         B          210
```

### agg() -- multiple aggregations at once

```python
result = sales.groupby("region")["revenue"].agg(["mean", "sum", "min", "max", "count"])
print(result)
#        mean  sum  min  max  count
# East  160.0  640  100  200      4
# West  122.5  490   90  150      4

# Different aggregations for different columns
result = sales.groupby("region").agg(
    total_revenue=("revenue", "sum"),
    avg_quantity=("quantity", "mean"),
    num_transactions=("revenue", "count"),
)
print(result)
```

### Custom aggregation functions

```python
# Using a lambda
result = sales.groupby("region")["revenue"].agg(
    lambda x: x.max() - x.min()
)
print("Revenue range by region:\n", result)

# Named aggregation with custom functions
def coefficient_of_variation(x):
    return x.std() / x.mean() * 100

result = sales.groupby("region")["revenue"].agg(
    mean="mean",
    cv=coefficient_of_variation,
)
print(result)
```

### transform() -- group stats broadcast back to original shape

`transform` is essential for feature engineering: it computes a group-level statistic but returns a Series the same length as the input DataFrame.

```python
# Add "group mean" as a new column (same shape as original)
sales["region_avg_revenue"] = sales.groupby("region")["revenue"].transform("mean")
print(sales)
# Each row now has its region's average -- useful as a feature

# Normalize revenue within each region (z-score by group)
sales["revenue_zscore"] = sales.groupby("region")["revenue"].transform(
    lambda x: (x - x.mean()) / x.std()
)
print(sales[["region", "revenue", "revenue_zscore"]])

# Count of items per group (like SQL COUNT(*) OVER (PARTITION BY ...))
sales["region_count"] = sales.groupby("region")["revenue"].transform("count")
```

### filter() -- keep or drop entire groups

```python
# Keep only regions with total revenue > 500
high_revenue = sales.groupby("region").filter(lambda g: g["revenue"].sum() > 500)
print(high_revenue)
```

## Section 2: Merging and Joining DataFrames

### merge() -- SQL-style joins

```python
# Users table
users = pd.DataFrame({
    "user_id": [1, 2, 3, 4],
    "name": ["Alice", "Bob", "Carol", "Dave"],
    "plan": ["pro", "free", "pro", "enterprise"],
})

# Transactions table
transactions = pd.DataFrame({
    "tx_id": [101, 102, 103, 104, 105],
    "user_id": [1, 2, 1, 5, 3],
    "amount": [29.99, 0, 49.99, 19.99, 99.99],
    "product": ["API", "Trial", "API", "API", "Training"],
})
```

### Inner join (default) -- only matching rows

```python
inner = pd.merge(users, transactions, on="user_id", how="inner")
print(inner)
# user_id 5 is dropped (not in users)
# user_id 4 is dropped (not in transactions)
```

### Left join -- all rows from the left table

```python
left = pd.merge(users, transactions, on="user_id", how="left")
print(left)
# user_id 4 (Dave) appears with NaN for transaction columns
```

### Right join and outer join

```python
right = pd.merge(users, transactions, on="user_id", how="right")
print(right)
# user_id 5 appears with NaN for user columns

outer = pd.merge(users, transactions, on="user_id", how="outer")
print(outer)
# All rows from both tables, NaN where no match
```

### Merging on different column names

```python
# If the key columns have different names
df_a = pd.DataFrame({"emp_id": [1, 2], "name": ["Alice", "Bob"]})
df_b = pd.DataFrame({"employee_id": [1, 2], "salary": [100, 120]})

merged = pd.merge(df_a, df_b, left_on="emp_id", right_on="employee_id")
print(merged)
```

### concat() -- stacking DataFrames

```python
# Vertical concatenation (stacking rows)
jan = pd.DataFrame({"date": ["2024-01-01"], "revenue": [100]})
feb = pd.DataFrame({"date": ["2024-02-01"], "revenue": [120]})
mar = pd.DataFrame({"date": ["2024-03-01"], "revenue": [115]})

quarterly = pd.concat([jan, feb, mar], ignore_index=True)
print(quarterly)

# Horizontal concatenation (adding columns)
features = pd.DataFrame({"height": [170, 165], "weight": [70, 55]})
labels = pd.DataFrame({"target": [1, 0]})

dataset = pd.concat([features, labels], axis=1)
print(dataset)
```

### Merge indicator -- debugging joins

```python
result = pd.merge(users, transactions, on="user_id", how="outer", indicator=True)
print(result["_merge"].value_counts())
# both          3
# left_only     1
# right_only    1
# Very useful for finding unmatched rows
```

## Section 3: Pivot Tables, Melt, Stack/Unstack

### pivot_table() -- summarize data in a cross-tabulation

```python
sales = pd.DataFrame({
    "date": ["2024-01", "2024-01", "2024-02", "2024-02", "2024-01", "2024-02"],
    "region": ["East", "West", "East", "West", "East", "West"],
    "product": ["A", "A", "A", "A", "B", "B"],
    "revenue": [100, 150, 120, 130, 80, 90],
})

# Pivot: rows=region, columns=date, values=revenue
pivot = sales.pivot_table(
    values="revenue",
    index="region",
    columns="date",
    aggfunc="sum",
)
print(pivot)
#         2024-01  2024-02
# East      180      120
# West      150      220

# Multiple aggregations
pivot2 = sales.pivot_table(
    values="revenue",
    index="region",
    columns="product",
    aggfunc=["sum", "mean"],
    fill_value=0,       # fill missing combinations with 0
    margins=True,        # add row/column totals
)
print(pivot2)
```

### melt() -- wide to long format

`melt` is the inverse of pivot -- it "unpivots" a wide DataFrame into a long one.

```python
# Wide format: one column per exam
wide = pd.DataFrame({
    "student": ["Alice", "Bob", "Carol"],
    "math": [90, 85, 78],
    "science": [88, 92, 80],
    "english": [76, 81, 95],
})

# Melt to long format
long = wide.melt(
    id_vars=["student"],
    value_vars=["math", "science", "english"],
    var_name="subject",
    value_name="score",
)
print(long)
#   student  subject  score
# 0   Alice     math     90
# 1     Bob     math     85
# 2   Carol     math     78
# 3   Alice  science     88
# ...
```

### stack() and unstack() -- reshaping multi-level indices

```python
# Create a multi-index DataFrame
df = sales.groupby(["region", "product"])["revenue"].sum()
print(df)
# region  product
# East    A          220
#         B           80
# West    A          280
#         B           90

# Unstack -- pivot the inner index level to columns
print(df.unstack())
# product    A    B
# region
# East     220   80
# West     280   90

# Stack -- the reverse: columns to index level
wide_df = df.unstack()
print(wide_df.stack())
# Back to the original long format
```

## Section 4: Worked Example -- Feature Engineering for a Churn Model

```python
import pandas as pd
import numpy as np
from io import StringIO

# Customer demographics
customers_csv = """customer_id,name,signup_date,plan
C001,Alice,2023-01-15,premium
C002,Bob,2023-03-22,basic
C003,Carol,2022-11-01,premium
C004,Dave,2023-06-10,basic
C005,Eve,2023-02-28,enterprise
"""

# Transaction history
transactions_csv = """tx_id,customer_id,date,amount,category
T001,C001,2024-01-05,49.99,subscription
T002,C001,2024-01-12,12.50,addon
T003,C002,2024-01-08,9.99,subscription
T004,C003,2024-01-03,49.99,subscription
T005,C003,2024-01-15,25.00,addon
T006,C003,2024-01-22,15.00,addon
T007,C001,2024-02-05,49.99,subscription
T008,C005,2024-01-10,99.99,subscription
T009,C005,2024-02-10,99.99,subscription
T010,C002,2024-02-08,9.99,subscription
"""

customers = pd.read_csv(StringIO(customers_csv))
transactions = pd.read_csv(StringIO(transactions_csv))

# Step 1: Merge customers with transaction aggregates
tx_features = transactions.groupby("customer_id").agg(
    total_spend=("amount", "sum"),
    num_transactions=("amount", "count"),
    avg_transaction=("amount", "mean"),
    max_transaction=("amount", "max"),
    num_categories=("category", "nunique"),
).reset_index()

print("Transaction features:")
print(tx_features)

# Step 2: Merge with customer table
df = pd.merge(customers, tx_features, on="customer_id", how="left")

# Step 3: Fill NaN for customers with no transactions (Dave)
df["total_spend"] = df["total_spend"].fillna(0)
df["num_transactions"] = df["num_transactions"].fillna(0).astype(int)
df["avg_transaction"] = df["avg_transaction"].fillna(0)
df["max_transaction"] = df["max_transaction"].fillna(0)
df["num_categories"] = df["num_categories"].fillna(0).astype(int)

# Step 4: Add plan-level features using transform
df["plan_avg_spend"] = df.groupby("plan")["total_spend"].transform("mean")

# Step 5: Create a pivot showing spend by customer by category
spend_by_cat = transactions.pivot_table(
    values="amount",
    index="customer_id",
    columns="category",
    aggfunc="sum",
    fill_value=0,
)
print("\nSpend by category:")
print(spend_by_cat)

# Step 6: Merge pivot back
df = pd.merge(df, spend_by_cat, on="customer_id", how="left").fillna(0)

print("\nFinal feature matrix:")
print(df)
print(f"\nShape: {df.shape}")
```

## Common Mistakes

- **Forgetting `reset_index()` after groupby.** A grouped aggregation returns a Series or DataFrame with a multi-level index. Add `.reset_index()` to get a flat DataFrame suitable for merging.
- **Wrong join type causing data loss.** An inner join silently drops non-matching rows. Always check row counts before and after merging and use the `indicator=True` parameter to debug.
- **Duplicated rows after a merge.** If the join key is not unique in one or both tables, you get a many-to-many join that multiplies rows. Check for duplicates before merging.
- **Using `apply` when `transform` is needed.** `apply` returns one value per group (reducing the DataFrame). `transform` returns one value per row (preserving the original shape).
- **Confusing `pivot_table` and `pivot`.** `pivot` requires unique index/column pairs and does not aggregate. `pivot_table` handles duplicates by aggregating. Prefer `pivot_table` in most cases.

### Solution 1: GroupBy basics

```python
# Average salary by department
print(employees.groupby("department")["salary"].mean())

# Count of employees per department
print(employees.groupby("department")["emp_id"].count())

# Max and min salary per department in one agg call
print(employees.groupby("department")["salary"].agg(["max", "min"]))
```

### Solution 2: GroupBy with transform

```python
employees = employees.copy()
employees["dept_avg_salary"] = employees.groupby("department")["salary"].transform("mean")
employees["salary_vs_dept_avg"] = employees["salary"] - employees["dept_avg_salary"]
print(employees[["name", "department", "salary", "dept_avg_salary", "salary_vs_dept_avg"]])
```

### Solution 3: Merge

```python
# Inner join
inner = pd.merge(employees, projects, on="emp_id", how="inner")
print(f"Inner join rows: {len(inner)}")

# Left join -- includes employees with no projects
left = pd.merge(employees, projects, on="emp_id", how="left")
no_project = left[left["project_id"].isna()]["name"]

# With indicator column
left_ind = pd.merge(employees, projects, on="emp_id", how="left", indicator=True)
```

### Solution 4: Aggregation after merge

```python
# Total hours per employee
merged = pd.merge(employees, projects, on="emp_id", how="left")
hours_per_emp = merged.groupby(["emp_id", "name"])["hours"].sum().reset_index()
print("Total hours per employee:\n", hours_per_emp)

# Employee with most hours
top_emp = hours_per_emp.loc[hours_per_emp["hours"].idxmax()]
print(f"Most hours: {top_emp['name']} ({top_emp['hours']} hrs)")

# Average hours per department (via merged data)
merged_full = pd.merge(employees, projects, on="emp_id", how="inner")
dept_hours = merged_full.groupby("department")["hours"].mean()

```

### Solution 5: Pivot table

```python
# Pivot: department x project status -- count of projects
```
