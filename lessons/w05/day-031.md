## Section 1: Matrix Multiplication

### The three ways to multiply matrices

```python
import numpy as np

A = np.array([[1, 2],
              [3, 4]])

B = np.array([[5, 6],
              [7, 8]])

# Method 1: np.dot
C = np.dot(A, B)

# Method 2: np.matmul
C = np.matmul(A, B)

# Method 3: @ operator (preferred -- clean and readable)
C = A @ B

print(C)
# [[19 22]
#  [43 50]]
```

### Remember: `*` is element-wise, `@` is matrix multiplication

```python
print(A * B)    # [[ 5 12], [21 32]]  -- element-wise
print(A @ B)    # [[19 22], [43 50]]  -- matrix multiplication
```

### Dot product of vectors

```python
v1 = np.array([1, 2, 3])
v2 = np.array([4, 5, 6])

# Dot product: 1*4 + 2*5 + 3*6 = 32
print(np.dot(v1, v2))   # 32
print(v1 @ v2)           # 32
```

### Matrix-vector multiplication

This is the core operation in neural networks -- multiplying a weight matrix by an input vector.

```python
# Weight matrix (3 outputs, 4 inputs)
W = np.random.randn(3, 4)

# Input vector (4 features)
x = np.array([1.0, 0.5, -1.0, 2.0])

# Output = W @ x
output = W @ x
print(f"Input shape:  {x.shape}")     # (4,)
print(f"Weight shape: {W.shape}")     # (3, 4)
print(f"Output shape: {output.shape}")# (3,)
```

### Transpose

```python
A = np.array([[1, 2, 3],
              [4, 5, 6]])
print(A.shape)     # (2, 3)
print(A.T.shape)   # (3, 2)
print(A.T)
# [[1 4]
#  [2 5]
#  [3 6]]
```

## Section 2: np.linalg -- Linear Algebra Toolkit

### Inverse

```python
A = np.array([[1, 2],
              [3, 4]])

A_inv = np.linalg.inv(A)
print(A_inv)
# [[-2.   1. ]
#  [ 1.5 -0.5]]

# Verify: A @ A_inv should be the identity matrix
print(np.round(A @ A_inv, 10))
# [[1. 0.]
#  [0. 1.]]
```

### Determinant

```python
det = np.linalg.det(A)
print(f"Determinant: {det:.1f}")  # -2.0
# If det == 0, the matrix is singular (not invertible)
```

### Solving linear systems

Solve `Ax = b` for x -- more numerically stable than computing the inverse.

```python
# 2x + 3y = 8
# 4x + 1y = 6

A = np.array([[2, 3],
              [4, 1]])
b = np.array([8, 6])

x = np.linalg.solve(A, b)
print(x)   # [1. 2.]  meaning x=1, y=2

# Verify
print(A @ x)  # [8. 6.] -- matches b
```

### Eigenvalues and eigenvectors

Eigendecomposition is the foundation of PCA (Principal Component Analysis).

```python
# Symmetric matrix (covariance matrices are always symmetric)
C = np.array([[4, 2],
              [2, 3]])

eigenvalues, eigenvectors = np.linalg.eig(C)
print("Eigenvalues:", eigenvalues)    # [5. 2.]
print("Eigenvectors:\n", eigenvectors)
# Each column is an eigenvector

# Verify: C @ v = lambda * v
for i in range(len(eigenvalues)):
    v = eigenvectors[:, i]
    lam = eigenvalues[i]
    print(f"C @ v{i} = {C @ v}, lambda * v{i} = {lam * v}")
```

### Singular Value Decomposition (SVD)

SVD is used in dimensionality reduction, recommender systems, and data compression.

```python
# A matrix (e.g., user-item ratings)
M = np.array([[5, 4, 0, 1],
              [4, 0, 0, 1],
              [1, 1, 0, 5],
              [0, 0, 5, 4],
              [0, 1, 5, 4]])

U, sigma, Vt = np.linalg.svd(M, full_matrices=False)
print(f"U shape: {U.shape}")        # (5, 4)
print(f"Sigma: {sigma.round(2)}")   # singular values in descending order
print(f"Vt shape: {Vt.shape}")      # (4, 4)

# Reconstruct from top-k components (rank-k approximation)
k = 2
M_approx = U[:, :k] @ np.diag(sigma[:k]) @ Vt[:k, :]
print("Rank-2 approximation:\n", M_approx.round(1))
```

### Norms

```python
v = np.array([3, 4])
print(np.linalg.norm(v))           # 5.0  (L2 norm / Euclidean distance)
print(np.linalg.norm(v, ord=1))    # 7.0  (L1 norm / Manhattan distance)
print(np.linalg.norm(v, ord=np.inf))  # 4.0  (max absolute value)
```

## Section 3: Random Number Generation

Reproducible randomness is essential for ML experiments.

```python
# Set seed for reproducibility
np.random.seed(42)

# Uniform distribution [0, 1)
uniform = np.random.rand(3, 3)
print("Uniform:\n", uniform.round(3))

# Standard normal (mean=0, std=1)
normal = np.random.randn(3, 3)
print("Normal:\n", normal.round(3))

# Normal with specific mean and std
# e.g., initialise weights for a neural network layer
weights = np.random.normal(loc=0.0, scale=0.01, size=(4, 3))
print("Weights:\n", weights.round(4))

# Uniform with specific range
learning_rates = np.random.uniform(low=0.0001, high=0.01, size=5)
print("Learning rates:", learning_rates.round(5))

# Random integers
labels = np.random.randint(0, 10, size=20)  # 20 labels from 0-9
print("Labels:", labels)

# Random choice from an array
classes = np.array(["cat", "dog", "bird"])
samples = np.random.choice(classes, size=10, replace=True)
print("Samples:", samples)

# Shuffle an array in place
deck = np.arange(52)
np.random.shuffle(deck)
print("Shuffled:", deck[:10])  # first 10 cards

# Permutation -- returns a new shuffled array (does NOT mutate)
perm = np.random.permutation(10)
print("Permutation:", perm)
```

### The modern way: Generator API (NumPy >= 1.17)

```python
rng = np.random.default_rng(seed=42)
data = rng.normal(0, 1, size=(3, 3))
print("Generator API:\n", data.round(3))
# Preferred in new code -- thread-safe and statistically better
```

## Section 4: Descriptive Statistics

```python
data = np.array([[10, 20, 30],
                 [40, 50, 60],
                 [70, 80, 90]])

# Global statistics
print(np.mean(data))       # 50.0
print(np.std(data))        # ~25.82
print(np.var(data))        # ~666.67
print(np.median(data))     # 50.0
print(np.min(data))        # 10
print(np.max(data))        # 90
print(np.sum(data))        # 450

# Axis-specific statistics (critical in ML)
# axis=0: operate along rows (result has shape of a single row)
print("Column means:", np.mean(data, axis=0))  # [40. 50. 60.]

# axis=1: operate along columns (result has shape of a single column)
print("Row means:", np.mean(data, axis=1))     # [20. 50. 80.]
```

### Percentiles and quantiles

```python
scores = np.array([45, 67, 72, 78, 82, 85, 88, 91, 95, 99])

print(np.percentile(scores, 25))   # 25th percentile (Q1): 73.5
print(np.percentile(scores, 50))   # 50th percentile (median): 83.5
print(np.percentile(scores, 75))   # 75th percentile (Q3): 90.25
print(np.percentile(scores, 90))   # 90th percentile: 94.2
```

### argmin, argmax -- finding positions

```python
predictions = np.array([0.1, 0.05, 0.7, 0.05, 0.1])
print(np.argmax(predictions))   # 2 -- index of highest probability
print(np.argmin(predictions))   # 1 -- index of lowest probability

# In a 2-D array: which class has the highest score per sample?
logits = np.array([[2.1, 0.5, -1.0],
                   [-0.3, 3.2, 1.1],
                   [0.0, 0.0, 4.5]])
predicted_classes = np.argmax(logits, axis=1)
print(predicted_classes)  # [0 1 2]
```

### Cumulative operations

```python
data = np.array([1, 2, 3, 4, 5])
print(np.cumsum(data))    # [ 1  3  6 10 15]
print(np.cumprod(data))   # [  1   2   6  24 120]
```

## Section 5: Worked Example -- Simple Linear Regression with Normal Equations

Implement linear regression from scratch using NumPy linear algebra.

```python
import numpy as np

# Generate synthetic data: y = 3x + 7 + noise
np.random.seed(42)
n_samples = 50
X_raw = np.random.uniform(0, 10, size=n_samples)
noise = np.random.normal(0, 2, size=n_samples)
y = 3 * X_raw + 7 + noise

# Build the design matrix (add a column of 1s for the intercept)
X = np.column_stack([X_raw, np.ones(n_samples)])
print(f"Design matrix shape: {X.shape}")  # (50, 2)

# Normal equation: theta = (X^T X)^{-1} X^T y
XtX = X.T @ X                    # (2, 2)
Xty = X.T @ y                    # (2,)
theta = np.linalg.solve(XtX, Xty)  # more stable than inv(XtX) @ Xty

slope, intercept = theta
print(f"Estimated slope:     {slope:.3f}  (true: 3.0)")
print(f"Estimated intercept: {intercept:.3f}  (true: 7.0)")

# Make predictions
y_pred = X @ theta

# Compute metrics
residuals = y - y_pred
mse = np.mean(residuals ** 2)
rmse = np.sqrt(mse)
r_squared = 1 - np.sum(residuals ** 2) / np.sum((y - np.mean(y)) ** 2)

print(f"RMSE: {rmse:.3f}")
print(f"R-squared: {r_squared:.3f}")
```

## Common Mistakes

- **Using `np.dot` for matrices when `@` is clearer.** `np.dot` has confusing behavior for higher-dimensional arrays. Prefer `@` (which maps to `np.matmul`) for clarity.
- **Computing the inverse to solve a linear system.** `np.linalg.solve(A, b)` is faster and more numerically stable than `np.linalg.inv(A) @ b`.
- **Forgetting to set the random seed.** Without a seed, results change every run. Always set `np.random.seed()` or use `default_rng(seed)` for reproducible experiments.
- **Confusing `axis=0` and `axis=1`.** `axis=0` collapses rows (operates "down" the rows, giving one result per column). `axis=1` collapses columns (operates "across" columns, giving one result per row).
- **Using `np.random.randn` when you need specific parameters.** `randn` gives standard normal (mean=0, std=1). Use `np.random.normal(mean, std, size)` for custom distributions.
