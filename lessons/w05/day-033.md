## Section 1: Handling Missing Values (NaN)

Missing data is represented in Pandas by `NaN` (Not a Number) or `pd.NA` (newer nullable type). NaN values propagate silently through calculations -- one NaN in a column can ruin an entire aggregation if you are not careful.

```python
import pandas as pd
import numpy as np

df = pd.DataFrame({
    "name": ["Alice", "Bob", None, "Dave", "Eve"],
    "age": [28, np.nan, 42, 31, 27],
    "score": [85.5, 92.0, np.nan, np.nan, 88.7],
    "city": ["NYC", "LA", "Chicago", None, "Miami"],
})
print(df)
```

### Detecting missing values

```python
# Boolean mask of missing values
print(df.isna())

# Count missing values per column
print(df.isna().sum())
# name     1
# age      1
# score    2
# city     1

# Percentage missing
print((df.isna().sum() / len(df) * 100).round(1))

# Any row with at least one NaN?
print(df.isna().any(axis=1))
```

### dropna -- remove rows or columns with missing data

```python
# Drop rows with ANY missing value
clean = df.dropna()
print(f"Before: {len(df)} rows, After: {len(clean)} rows")
# Before: 5, After: 2

# Drop rows only if a specific column is missing
clean2 = df.dropna(subset=["score"])
print(f"After dropping missing scores: {len(clean2)} rows")  # 3

# Drop columns with any missing value
clean3 = df.dropna(axis=1)

# Drop rows where ALL values are missing
# (useful for completely empty rows from bad CSV imports)
clean4 = df.dropna(how="all")

# Threshold: keep rows with at least N non-null values
clean5 = df.dropna(thresh=3)  # keep rows with at least 3 valid values
```

### fillna -- replace missing values

```python
# Fill with a constant
df["city"].fillna("Unknown", inplace=False)  # returns new Series

# Fill numeric with mean (very common in ML)
df["age"] = df["age"].fillna(df["age"].mean())

# Fill with median (more robust to outliers)
df["score"] = df["score"].fillna(df["score"].median())

# Forward fill -- carry the last valid value forward (time-series)
ts_data = pd.Series([1.0, np.nan, np.nan, 4.0, np.nan, 6.0])
print(ts_data.ffill())   # [1.0, 1.0, 1.0, 4.0, 4.0, 6.0]

# Backward fill
print(ts_data.bfill())   # [1.0, 4.0, 4.0, 4.0, 6.0, 6.0]
```

### interpolate -- smart filling for numeric/time data

```python
ts = pd.Series([0, np.nan, np.nan, 30, np.nan, 50])
print(ts.interpolate())
# 0     0.0
# 1    10.0
# 2    20.0
# 3    30.0
# 4    40.0
# 5    50.0
# Linear interpolation fills in the gaps evenly
```

## Section 2: Type Conversion

Mixed or wrong types are a constant headache. A column of numbers might be read as strings because one row has "N/A" in it.

### astype -- explicit type conversion

```python
df = pd.DataFrame({
    "user_id": ["001", "002", "003"],
    "age": ["25", "30", "28"],
    "score": [85, 92, 78],
})
print(df.dtypes)
# user_id    object  (string)
# age        object  (string!)
# score       int64

# Convert age to integer
df["age"] = df["age"].astype(int)

# Convert score to float32 (save memory)
df["score"] = df["score"].astype(np.float32)

print(df.dtypes)
```

### pd.to_numeric -- handles errors gracefully

```python
messy = pd.Series(["10", "20", "thirty", "40", "N/A"])

# This would crash: messy.astype(int)

# errors='coerce' turns unparseable values into NaN
clean = pd.to_numeric(messy, errors="coerce")
print(clean)
# 0    10.0
# 1    20.0
# 2     NaN
# 3    40.0
# 4     NaN
```

### pd.to_datetime -- parse dates

```python
dates = pd.Series(["2024-01-15", "2024-02-20", "March 3, 2024", "invalid"])

parsed = pd.to_datetime(dates, errors="coerce")
print(parsed)
# 0   2024-01-15
# 1   2024-02-20
# 2   2024-03-03
# 3          NaT    (Not a Time -- datetime's equivalent of NaN)

# Extract components
df_dates = pd.DataFrame({"date": parsed.dropna()})
df_dates["year"] = df_dates["date"].dt.year
df_dates["month"] = df_dates["date"].dt.month
df_dates["day_of_week"] = df_dates["date"].dt.day_name()
print(df_dates)
```

### Categorical type -- saves memory for repeated strings

```python
df = pd.DataFrame({
    "color": ["red", "blue", "red", "green", "blue"] * 1000,
})

print(f"Object dtype:  {df['color'].nbytes:,} bytes")  # ~40,000

df["color"] = df["color"].astype("category")
print(f"Category dtype: {df['color'].nbytes:,} bytes")  # ~5,000
# Huge memory savings for columns with few unique values
```

## Section 3: Handling Duplicates

```python
df = pd.DataFrame({
    "order_id": [101, 102, 103, 102, 104, 103],
    "product": ["A", "B", "C", "B", "D", "C"],
    "amount": [25.0, 30.0, 15.0, 30.0, 40.0, 15.0],
})

# Detect duplicates
print(df.duplicated())
# 0    False
# 1    False
# 2    False
# 3     True   (duplicate of row 1)
# 4    False
# 5     True   (duplicate of row 2)

# Count duplicates
print(f"Duplicates: {df.duplicated().sum()}")  # 2

# Check for duplicates based on specific columns
print(df.duplicated(subset=["order_id"]).sum())  # 2

# Remove duplicates
df_clean = df.drop_duplicates()
print(f"After drop_duplicates: {len(df_clean)} rows")  # 4

# Keep the last occurrence instead of the first
df_last = df.drop_duplicates(subset=["order_id"], keep="last")

# Drop ALL occurrences of duplicated rows
df_no_dupes = df.drop_duplicates(subset=["order_id"], keep=False)
print(f"After dropping all duplicated: {len(df_no_dupes)} rows")  # 2
```

## Section 4: String Methods with .str

The `.str` accessor gives you vectorized string operations -- no loops needed.

```python
df = pd.DataFrame({
    "name": ["  Alice Smith  ", "BOB JONES", "carol davis", "Dave  Brown"],
    "email": ["alice@GMAIL.com", "bob@yahoo.COM", "carol@gmail.com", None],
    "phone": ["123-456-7890", "(123) 456 7890", "1234567890", "123.456.7890"],
})
```

### Cleaning whitespace and case

```python
# Strip leading/trailing whitespace
df["name"] = df["name"].str.strip()

# Normalize case
df["name_lower"] = df["name"].str.lower()
df["name_title"] = df["name"].str.title()
df["email"] = df["email"].str.lower()  # NaN values are safely skipped
```

### Searching and extracting

```python
# Contains (returns boolean mask)
gmail_users = df[df["email"].str.contains("gmail", na=False)]
print(gmail_users)

# startswith / endswith
print(df["name"].str.startswith("A"))   # [True, False, False, False]

# Extract parts with split
df["first_name"] = df["name"].str.split().str[0]
df["last_name"] = df["name"].str.split().str[-1]
print(df[["first_name", "last_name"]])
```

### Replacing patterns

```python
# Simple replace
df["phone_clean"] = df["phone"].str.replace(r"[^0-9]", "", regex=True)
print(df["phone_clean"])
# 0    1234567890
# 1    1234567890
# 2    1234567890
# 3    1234567890

# Replace substrings
df["email"] = df["email"].str.replace("yahoo", "gmail", regex=False)
```

### String length and padding

```python
print(df["name"].str.len())

# Pad to a fixed width (useful for ID formatting)
ids = pd.Series([1, 23, 456])
print(ids.astype(str).str.zfill(5))
# 0    00001
# 1    00023
# 2    00456
```

## Section 5: Worked Example -- Cleaning a Real-World ML Dataset

```python
import pandas as pd
import numpy as np
from io import StringIO

raw_csv = """id,text,rating,timestamp,price
1,"  Great product! Would buy again.  ",5,2024-01-15 10:30:00,$29.99
2,"Terrible quality, broke immediately",1,2024-01-16 14:22:00,$15.50
3,,3,invalid_date,$N/A
4,"  AMAZING DEAL!!!  ",5,2024-02-01 09:00:00,$22.00
5,"decent, nothing special",3,2024-02-03 16:45:00,
6,"  Great product! Would buy again.  ",5,2024-01-15 10:30:00,$29.99
7,"ok",2,2024-03-01 08:00:00,$9.99
"""

df = pd.read_csv(StringIO(raw_csv))
print("=== Raw data ===")
print(df)
print(f"\nShape: {df.shape}")
print(f"\nMissing values:\n{df.isna().sum()}")

# Step 1: Remove duplicates
df = df.drop_duplicates()
print(f"\nAfter dedup: {len(df)} rows")

# Step 2: Clean text column
df["text"] = df["text"].str.strip().str.lower()
# Fill missing text with a placeholder
df["text"] = df["text"].fillna("[no text]")

# Step 3: Parse price -- remove $ sign, convert to float
df["price"] = df["price"].str.replace("$", "", regex=False)
df["price"] = pd.to_numeric(df["price"], errors="coerce")
# Fill missing prices with median
df["price"] = df["price"].fillna(df["price"].median())

# Step 4: Parse timestamps
df["timestamp"] = pd.to_datetime(df["timestamp"], errors="coerce")
# Fill invalid dates with the median date
median_date = df["timestamp"].dropna().median()
df["timestamp"] = df["timestamp"].fillna(median_date)

# Step 5: Validate rating range
df["rating"] = df["rating"].clip(1, 5)  # ensure 1-5 range

# Step 6: Add derived features
df["text_length"] = df["text"].str.len()
df["is_positive"] = df["rating"] >= 4

print("\n=== Cleaned data ===")
print(df)
print(f"\nDtypes:\n{df.dtypes}")
print(f"\nMissing values:\n{df.isna().sum()}")
```

## Common Mistakes

- **Filling NaN before understanding why it is missing.** Is the data missing at random, or is the missingness itself informative? Blindly filling with the mean can introduce bias.
- **Using `inplace=True` everywhere.** It makes code harder to debug (no intermediate results) and is being deprecated for many methods. Prefer reassignment: `df = df.dropna()`.
- **Forgetting `na=False` in `.str.contains()`.** If a column has NaN values, `.str.contains("pattern")` returns NaN for those rows, which causes errors in boolean indexing.
- **Casting to int when NaN values are present.** Integer columns cannot hold NaN in standard NumPy dtypes. Either drop/fill NaN first, or use the nullable `Int64` dtype: `df["col"].astype("Int64")`.
- **Over-cleaning.** Removing too many rows or filling too aggressively can reduce your dataset size or introduce bias. Always document your cleaning decisions.

```python
import pandas as pd
import numpy as np
from io import StringIO

messy_data = """student_id,name,grade,email,enrollment_date,gpa
STU001,"  John Smith ",A,john@university.EDU,2023-09-01,3.8
STU002,"jane DOE",B+,jane@university.edu,Sept 15 2023,3.5
STU003,"Bob Wilson",,bob@GMAIL.com,2023-09-01,N/A
STU001,"  John Smith ",A,john@university.EDU,2023-09-01,3.8
STU004,"Alice Brown",A-,alice@university.edu,invalid,3.9
STU005," CAROL  DAVIS",C,carol university edu,2023-10-01,2.1
STU006,"",B,dave@university.edu,2023-09-15,3.2
STU007,"frank miller",A,frank@university.edu,2023-09-01,
"""

df = pd.read_csv(StringIO(messy_data))
```

### Solution 1: Inspect

```python
print("Shape:", df.shape)
print("\nDtypes:\n", df.dtypes)
df.info()
print("\nMissing values:\n", df.isna().sum())
```

### Solution 2: Remove duplicates

```python
before = len(df)
df = df.drop_duplicates()
after = len(df)
print(f"Removed {before - after} duplicate row(s). Remaining: {after}")
```

### Solution 3: Clean names

```python
df["name"] = df["name"].str.strip().str.title()
# Replace empty strings with NaN, then fill with "Unknown"
df["name"] = df["name"].replace("", np.nan)
df["name"] = df["name"].fillna("Unknown")
print(df["name"])
```

### Solution 4: Clean emails

```python
df["email"] = df["email"].str.lower()
# Mark emails without "@" as invalid (NaN)
invalid_mask = ~df["email"].str.contains("@", na=False)
print("Invalid emails:\n", df.loc[invalid_mask, "email"])
df.loc[invalid_mask, "email"] = np.nan
```

### Solution 5: Fix GPA

```python
df["gpa"] = pd.to_numeric(df["gpa"], errors="coerce")
median_gpa = df["gpa"].median()
df["gpa"] = df["gpa"].fillna(median_gpa)
print(f"GPA median used for fill: {median_gpa:.2f}")
```

### Solution 6: Parse dates

```python
df["enrollment_date"] = pd.to_datetime(df["enrollment_date"], errors="coerce")
# Most common valid date
mode_date = df["enrollment_date"].dropna().mode()[0]
df["enrollment_date"] = df["enrollment_date"].fillna(mode_date)
print("Enrollment dates:\n", df["enrollment_date"])
```

### Solution 7: Validate grades

```python
print("Unique grades:", df["grade"].unique())
df["grade"] = df["grade"].replace("", np.nan)
df["grade"] = df["grade"].fillna("incomplete")
print("After fix:", df["grade"].unique())
```

### Solution 8: Final check

```python
print("\n=== Cleaned DataFrame ===")
print(df)
print("\nMissing values:\n", df.isna().sum())
print("\nDtypes:\n", df.dtypes)
```
