## Section 1: The GIL and Why It Matters

Python's Global Interpreter Lock (GIL) allows only **one thread** to execute Python
bytecode at a time. This means:

| Task type | Threads help? | Processes help? | Async helps? |
|-----------|---------------|-----------------|--------------|
| I/O-bound (network, disk) | Yes | Yes | Yes |
| CPU-bound (math, training) | No | Yes | No |

- **I/O-bound:** The thread releases the GIL while waiting for the OS to finish the I/O
  operation. Other threads can run during that wait.
- **CPU-bound:** The GIL means only one thread crunches numbers at a time. You need
  separate **processes** (each with its own GIL) for true parallelism.

## Section 2: Threading

### Basic Thread

```python
import threading
import time

def download_data(url: str, delay: float = 1.0):
    """Simulate downloading data."""
    print(f"[{threading.current_thread().name}] Starting download: {url}")
    time.sleep(delay)  # simulates network wait
    print(f"[{threading.current_thread().name}] Finished: {url}")

# Sequential: ~3 seconds
urls = ["data/train.csv", "data/val.csv", "data/test.csv"]
start = time.perf_counter()
for url in urls:
    download_data(url)
print(f"Sequential: {time.perf_counter() - start:.2f}s\n")

# Threaded: ~1 second
start = time.perf_counter()
threads = []
for url in urls:
    t = threading.Thread(target=download_data, args=(url,))
    threads.append(t)
    t.start()

for t in threads:
    t.join()  # wait for all threads to complete

print(f"Threaded: {time.perf_counter() - start:.2f}s")
```

### ThreadPoolExecutor (preferred)

The `concurrent.futures` module provides a cleaner, higher-level API.

```python
from concurrent.futures import ThreadPoolExecutor, as_completed
import time

def fetch_prediction(model_id: int) -> dict:
    """Simulate calling a model API."""
    time.sleep(0.5)  # simulates network latency
    return {"model_id": model_id, "score": 0.85 + model_id * 0.01}

model_ids = list(range(1, 11))

start = time.perf_counter()
results = []

with ThreadPoolExecutor(max_workers=5) as executor:
    # Submit all tasks
    future_to_id = {
        executor.submit(fetch_prediction, mid): mid
        for mid in model_ids
    }

    # Collect results as they complete
    for future in as_completed(future_to_id):
        mid = future_to_id[future]
        try:
            result = future.result()
            results.append(result)
        except Exception as e:
            print(f"Model {mid} failed: {e}")

elapsed = time.perf_counter() - start
print(f"Fetched {len(results)} predictions in {elapsed:.2f}s")
# ~1.0s instead of ~5.0s sequential
```

### Using executor.map for simpler cases

```python
from concurrent.futures import ThreadPoolExecutor
import time

def process_item(item: int) -> int:
    time.sleep(0.3)
    return item ** 2

items = list(range(20))

with ThreadPoolExecutor(max_workers=8) as executor:
    results = list(executor.map(process_item, items))

print(results)  # [0, 1, 4, 9, 16, ...]
```

## Section 3: Multiprocessing

For CPU-bound work, you need separate processes to bypass the GIL.

### Basic Process

```python
import multiprocessing
import time
import math

def heavy_computation(n: int) -> float:
    """CPU-intensive task: compute sum of square roots."""
    return sum(math.sqrt(i) for i in range(n))

# Sequential
start = time.perf_counter()
results = [heavy_computation(5_000_000) for _ in range(4)]
print(f"Sequential: {time.perf_counter() - start:.2f}s")

# Parallel with processes
start = time.perf_counter()
with multiprocessing.Pool(processes=4) as pool:
    results = pool.map(heavy_computation, [5_000_000] * 4)
print(f"Parallel: {time.perf_counter() - start:.2f}s")
```

### ProcessPoolExecutor

Same API as ThreadPoolExecutor -- just swap the class.

```python
from concurrent.futures import ProcessPoolExecutor
import math
import time

def cpu_task(n: int) -> float:
    """Simulate CPU-heavy ML preprocessing."""
    total = 0.0
    for i in range(n):
        total += math.sin(i) * math.cos(i)
    return total

if __name__ == "__main__":
    data_chunks = [2_000_000] * 8

    start = time.perf_counter()
    with ProcessPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(cpu_task, data_chunks))
    elapsed = time.perf_counter() - start

    print(f"Processed {len(results)} chunks in {elapsed:.2f}s")
```

> **Important:** On macOS and Windows, multiprocessing code must be inside
> `if __name__ == "__main__":` to avoid infinite spawn loops.

### When to use which

```
Need to call 100 APIs?       -> ThreadPoolExecutor or asyncio
Need to preprocess 1M images? -> ProcessPoolExecutor
Need to train a model?        -> Use the framework's built-in parallelism (PyTorch DataLoader)
Need to download + process?   -> Combine: threads for download, processes for compute
```

## Section 4: asyncio and async/await

Asyncio uses a single thread with cooperative multitasking. Instead of blocking on I/O,
a coroutine **yields** control back to the event loop, letting other coroutines run.

### Basics

```python
import asyncio

async def fetch_data(name: str, delay: float) -> str:
    """Simulate an async API call."""
    print(f"Starting {name}...")
    await asyncio.sleep(delay)  # non-blocking sleep
    print(f"Finished {name}")
    return f"{name}: done"

async def main():
    # Run three tasks concurrently
    results = await asyncio.gather(
        fetch_data("dataset_a", 2.0),
        fetch_data("dataset_b", 1.5),
        fetch_data("dataset_c", 1.0),
    )
    print(results)

# In a script:
asyncio.run(main())

# In Jupyter, the event loop is already running, so use:
# await main()
```

All three tasks run concurrently in **one thread**. Total time is ~2.0s (the longest),
not 4.5s (the sum).

### Creating tasks

```python
async def main():
    # create_task schedules the coroutine immediately
    task1 = asyncio.create_task(fetch_data("a", 2.0))
    task2 = asyncio.create_task(fetch_data("b", 1.0))

    # Do other work while tasks run...
    print("Tasks are running in the background")

    # Await results when you need them
    result1 = await task1
    result2 = await task2
    print(result1, result2)

asyncio.run(main())
```

### Async with aiohttp

`requests` is synchronous. For async HTTP, use `aiohttp`.

```python
# pip install aiohttp
import aiohttp
import asyncio

async def fetch_url(session: aiohttp.ClientSession, url: str) -> dict:
    """Fetch a URL asynchronously."""
    async with session.get(url) as response:
        return await response.json()

async def main():
    urls = [
        f"https://jsonplaceholder.typicode.com/posts/{i}"
        for i in range(1, 11)
    ]

    async with aiohttp.ClientSession() as session:
        tasks = [fetch_url(session, url) for url in urls]
        results = await asyncio.gather(*tasks)

    for r in results:
        print(f"Post {r['id']}: {r['title'][:40]}...")

asyncio.run(main())
```

### Async with timeouts

```python
async def fetch_with_timeout(url: str, timeout_sec: float = 5.0) -> dict:
    """Fetch with a per-request timeout."""
    timeout = aiohttp.ClientTimeout(total=timeout_sec)
    async with aiohttp.ClientSession(timeout=timeout) as session:
        try:
            async with session.get(url) as response:
                return await response.json()
        except asyncio.TimeoutError:
            print(f"Timeout fetching {url}")
            return {}
```

### Semaphore for rate limiting

```python
async def fetch_limited(sem: asyncio.Semaphore, session, url: str) -> dict:
    """Fetch with concurrency limit."""
    async with sem:  # only N coroutines enter at a time
        async with session.get(url) as response:
            return await response.json()

async def main():
    sem = asyncio.Semaphore(5)  # max 5 concurrent requests
    urls = [f"https://jsonplaceholder.typicode.com/posts/{i}" for i in range(1, 51)]

    async with aiohttp.ClientSession() as session:
        tasks = [fetch_limited(sem, session, url) for url in urls]
        results = await asyncio.gather(*tasks)

    print(f"Fetched {len(results)} posts")

asyncio.run(main())
```

## Section 5: Practical Patterns for AI Engineering

### Parallel data loading

```python
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import json

def load_json_file(path: Path) -> dict:
    """Load a single JSON file."""
    with open(path) as f:
        return json.load(f)

def load_all_json(directory: str, max_workers: int = 8) -> list[dict]:
    """Load all JSON files from a directory in parallel."""
    paths = list(Path(directory).glob("*.json"))

    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        results = list(executor.map(load_json_file, paths))

    print(f"Loaded {len(results)} files")
    return results
```

### Concurrent model inference

```python
import asyncio
import aiohttp

async def infer_batch(texts: list[str], api_url: str, api_key: str) -> list[dict]:
    """Send multiple texts to an inference API concurrently."""
    sem = asyncio.Semaphore(10)

    async def infer_one(session, text):
        async with sem:
            async with session.post(
                api_url,
                json={"inputs": text},
                headers={"Authorization": f"Bearer {api_key}"}
            ) as resp:
                return await resp.json()

    async with aiohttp.ClientSession() as session:
        tasks = [infer_one(session, text) for text in texts]
        return await asyncio.gather(*tasks)

# Usage (pseudocode -- needs real endpoint):
# results = asyncio.run(infer_batch(texts, API_URL, API_KEY))
```

### Mixed approach: threads + processes

```python
from concurrent.futures import ThreadPoolExecutor, ProcessPoolExecutor
import time

def download(url: str) -> bytes:
    """I/O-bound: download data."""
    time.sleep(0.5)  # simulate
    return b"raw data"

def preprocess(data: bytes) -> list:
    """CPU-bound: heavy processing."""
    time.sleep(1.0)  # simulate
    return [1, 2, 3]

def pipeline(urls: list[str]) -> list:
    # Step 1: Download in parallel (I/O-bound -> threads)
    with ThreadPoolExecutor(max_workers=8) as executor:
        raw_data = list(executor.map(download, urls))

    # Step 2: Preprocess in parallel (CPU-bound -> processes)
    with ProcessPoolExecutor(max_workers=4) as executor:
        processed = list(executor.map(preprocess, raw_data))

    return processed

if __name__ == "__main__":
    urls = [f"https://data.example.com/chunk_{i}" for i in range(12)]
    start = time.perf_counter()
    results = pipeline(urls)
    print(f"Pipeline: {time.perf_counter() - start:.2f}s, {len(results)} items")
```

## Common Mistakes

| Mistake | Fix |
|---------|-----|
| Using threads for CPU-bound work | Use `ProcessPoolExecutor` instead |
| Forgetting `if __name__ == "__main__"` with multiprocessing | Always guard multiprocessing entry points |
| Calling `asyncio.run()` inside Jupyter | Use `await main()` directly; the event loop is already running |
| Creating too many threads/processes | Match `max_workers` to your task; too many adds overhead |
| Sharing mutable state between threads without locks | Use `threading.Lock()` or avoid shared state |
| Using `time.sleep()` in async code | Use `await asyncio.sleep()` -- `time.sleep` blocks the event loop |

## Check Yourself

1. What is the GIL and how does it affect threads vs processes?
2. When should you use `ThreadPoolExecutor` vs `ProcessPoolExecutor`?
3. What does `await asyncio.gather(*tasks)` do?
4. Why does `aiohttp` exist when we already have `requests`?
5. How does `asyncio.Semaphore` help with rate limiting?

<details>
<summary>Answers</summary>

1. The GIL (Global Interpreter Lock) allows only one thread to execute Python bytecode
   at a time. Threads help for I/O-bound work (they release the GIL during waits),
   but not for CPU-bound work. Processes each have their own GIL, so they achieve
   true parallelism.
2. `ThreadPoolExecutor` for I/O-bound tasks (API calls, file I/O, database queries).
   `ProcessPoolExecutor` for CPU-bound tasks (data preprocessing, numerical computation).
3. It runs all the given coroutines concurrently and returns a list of their results
   in the same order, waiting until all are complete.
4. `requests` is synchronous -- each call blocks until it completes. `aiohttp` is
   built for asyncio, allowing hundreds of concurrent HTTP requests in a single thread.
5. A `Semaphore(n)` ensures that at most `n` coroutines can enter a critical section
   at the same time. This prevents overwhelming an API with too many concurrent requests.

</details>

### Full solution (all parts together)

```python
import requests
import time
import asyncio
import aiohttp
from concurrent.futures import ThreadPoolExecutor, ProcessPoolExecutor, as_completed

BASE_URL = "https://jsonplaceholder.typicode.com/posts"
POST_IDS = list(range(1, 51))


# --- I/O: sequential ---
def fetch_sequential(ids):
    results = []
    for pid in ids:
        r = requests.get(f"{BASE_URL}/{pid}", timeout=10)
        r.raise_for_status()
        results.append(r.json())
    return results


# --- I/O: threaded ---
def fetch_one(pid):
    r = requests.get(f"{BASE_URL}/{pid}", timeout=10)
    r.raise_for_status()
    return r.json()

def fetch_threaded(ids, workers=10):
    with ThreadPoolExecutor(max_workers=workers) as ex:
        futures = {ex.submit(fetch_one, pid): pid for pid in ids}
        results = []
        for future in as_completed(futures):
            try:
                results.append(future.result())
            except Exception as e:
                print(f"Thread error: {e}")
    return results


# --- I/O: async ---
async def _fetch_async(post_ids, concurrency=10):
    sem = asyncio.Semaphore(concurrency)

    async def one(session, pid):
        async with sem:
            async with session.get(f"{BASE_URL}/{pid}") as resp:
                return await resp.json()

    async with aiohttp.ClientSession() as session:
        return await asyncio.gather(*[one(session, pid) for pid in post_ids])

def fetch_async_run(ids):
    return asyncio.run(_fetch_async(ids))


# --- CPU ---
def sum_of_squares(n):
    return sum(i * i for i in range(n))

def cpu_sequential(runs):
    return [sum_of_squares(n) for n in runs]

def cpu_parallel(runs, workers=4):
    with ProcessPoolExecutor(max_workers=workers) as ex:
        return list(ex.map(sum_of_squares, runs))


def timed(fn, *args):
    start = time.perf_counter()
    result = fn(*args)
    return result, time.perf_counter() - start


if __name__ == "__main__":
    N, cpu_runs = 10_000_000, [10_000_000] * 4

    _, t_seq    = timed(fetch_sequential, POST_IDS)
    _, t_thread = timed(fetch_threaded,   POST_IDS)
    _, t_async  = timed(fetch_async_run,  POST_IDS)
    _, t_cpu_s  = timed(cpu_sequential,   cpu_runs)
    _, t_cpu_p  = timed(cpu_parallel,     cpu_runs)

    rows = [
        ("Sequential HTTP",  t_seq,    1.0),
        ("ThreadPool HTTP",  t_thread, t_seq   / t_thread),
        ("Async HTTP",       t_async,  t_seq   / t_async),
        ("Sequential CPU",   t_cpu_s,  1.0),
        ("ProcessPool CPU",  t_cpu_p,  t_cpu_s / t_cpu_p),
    ]
    print(f"\n{'Method':<25} {'Time (s)':>10} {'Speedup':>10}")
    print("-" * 48)
    for name, elapsed, speedup in rows:
        print(f"{name:<25} {elapsed:>10.2f} {speedup:>9.1f}x")
```
