## Section 1: HTTP Basics

Every REST API interaction is an HTTP request-response cycle.

| Method  | Purpose                    | Has Body? |
|---------|----------------------------|-----------|
| GET     | Retrieve data              | No        |
| POST    | Create a new resource      | Yes       |
| PUT     | Replace an existing resource | Yes     |
| PATCH   | Partially update a resource | Yes      |
| DELETE  | Remove a resource          | No        |

### Status code families

| Range | Meaning      | Common codes                      |
|-------|--------------|-----------------------------------|
| 2xx   | Success      | 200 OK, 201 Created, 204 No Content |
| 3xx   | Redirect     | 301, 304                          |
| 4xx   | Client error | 400 Bad Request, 401 Unauthorized, 403 Forbidden, 404 Not Found, 429 Too Many Requests |
| 5xx   | Server error | 500 Internal Server Error, 503 Service Unavailable |

## Section 2: The requests Library

```python
import requests

# Basic GET request
response = requests.get("https://jsonplaceholder.typicode.com/posts/1")

print(response.status_code)    # 200
print(response.headers["Content-Type"])  # application/json; charset=utf-8
print(response.json())         # parsed JSON as a dict
```

### GET with query parameters

```python
# These two are equivalent:
# requests.get("https://api.example.com/search?q=python&limit=10")
response = requests.get(
    "https://jsonplaceholder.typicode.com/posts",
    params={"userId": 1}
)

print(f"Found {len(response.json())} posts for userId=1")

# Always check the status
response.raise_for_status()  # raises HTTPError if 4xx/5xx
```

### POST request

```python
new_post = {
    "title": "My ML Experiment Results",
    "body": "Accuracy: 0.95, F1: 0.93",
    "userId": 1
}

response = requests.post(
    "https://jsonplaceholder.typicode.com/posts",
    json=new_post   # automatically sets Content-Type: application/json

```

### PUT and DELETE


## Section 3: Authentication Patterns

### API key in header

```python
import os

API_KEY = os.environ.get("MY_API_KEY", "demo-key-123")

response = requests.get(
    "https://api.example.com/data",
    headers={"X-API-Key": API_KEY}
)
```

### Bearer token (OAuth2)

```python
TOKEN = os.environ.get("AUTH_TOKEN", "your-token-here")

response = requests.get(
    "https://api.example.com/protected",
    headers={"Authorization": f"Bearer {TOKEN}"}
)
```

### Using a Session for repeated requests

When you make many requests to the same API, a `Session` reuses the TCP connection
and keeps headers across requests.

```python
session = requests.Session()
session.headers.update({
    "Authorization": f"Bearer {TOKEN}",
    "Accept": "application/json"
})

# All requests through this session carry the auth header
r1 = session.get("https://api.example.com/users")
r2 = session.get("https://api.example.com/models")

session.close()
```

### Pattern: API key from environment

```python
import os
import sys

def get_api_key(env_var: str) -> str:
    """Safely retrieve an API key from environment variables."""
    key = os.environ.get(env_var)
    if key is None:
        print(f"Error: {env_var} not set. Export it first:")
        print(f"  export {env_var}='your-key-here'")
        sys.exit(1)
    return key
```

## Section 4: Error Handling

### Basic pattern

```python
import requests
from requests.exceptions import HTTPError, ConnectionError, Timeout

def fetch_data(url: str, params: dict = None) -> dict:
    """Fetch JSON data with proper error handling."""
    try:
        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()
        return response.json()
    except Timeout:
        print(f"Request timed out: {url}")
        return {}
    except ConnectionError:
        print(f"Could not connect to: {url}")
        return {}
    except HTTPError as e:
        print(f"HTTP error {e.response.status_code}: {e}")
        return {}

data = fetch_data("https://jsonplaceholder.typicode.com/posts/1")
print(data.get("title", "No data"))
```

### Retry with exponential backoff

```python
import time
import requests
from requests.exceptions import HTTPError

def fetch_with_retry(url: str, max_retries: int = 3, backoff: float = 1.0) -> dict:
    """GET request with exponential backoff retries."""
    for attempt in range(max_retries):
        try:
            response = requests.get(url, timeout=10)
            response.raise_for_status()
            return response.json()
        except (HTTPError, requests.exceptions.ConnectionError) as e:
            wait = backoff * (2 ** attempt)
            print(f"Attempt {attempt + 1} failed: {e}")
            if attempt < max_retries - 1:
                print(f"Retrying in {wait:.1f}s...")
                time.sleep(wait)
            else:
                print("All retries exhausted.")
                raise
    return {}
```

### Using requests adapters for automatic retries

```python
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

def create_retry_session(
    retries: int = 3,
    backoff_factor: float = 0.5,
    status_forcelist: tuple = (500, 502, 503, 504)
) -> requests.Session:
    """Create a session with automatic retry on failure."""
    session = requests.Session()
    retry = Retry(
        total=retries,
        backoff_factor=backoff_factor,
        status_forcelist=status_forcelist,
        allowed_methods=["GET", "POST"]
    )
    adapter = HTTPAdapter(max_retries=retry)
    session.mount("http://", adapter)
    session.mount("https://", adapter)
    return session

# Usage
session = create_retry_session()
response = session.get("https://jsonplaceholder.typicode.com/posts/1")
print(response.json()["title"])
```

## Section 5: Pagination

Most APIs return data in pages. There are several common patterns.

### Offset-based pagination

```python
def fetch_all_offset(base_url: str, page_size: int = 20) -> list:
    """Collect all items using offset pagination."""
    all_items = []
    offset = 0

    while True:
        response = requests.get(
            base_url,
            params={"offset": offset, "limit": page_size},
            timeout=10
        )
        response.raise_for_status()
        data = response.json()

        items = data.get("results", [])
        if not items:
            break

        all_items.extend(items)
        offset += page_size
        print(f"Fetched {len(all_items)} items so far...")

    return all_items
```

### Cursor-based pagination

```python
def fetch_all_cursor(base_url: str) -> list:
    """Collect all items using cursor-based pagination."""
    all_items = []
    cursor = None

    while True:
        params = {}
        if cursor:
            params["cursor"] = cursor

        response = requests.get(base_url, params=params, timeout=10)
        response.raise_for_status()
        data = response.json()

        all_items.extend(data.get("items", []))
        cursor = data.get("next_cursor")

        if not cursor:
            break

        print(f"Fetched {len(all_items)} items, next cursor: {cursor[:20]}...")

    return all_items
```

### Link-header pagination (GitHub style)

```python
def fetch_all_link_header(url: str) -> list:
    """Collect all items using Link header pagination (GitHub API style)."""
    all_items = []

    while url:
        response = requests.get(url, timeout=10)
        response.raise_for_status()
        all_items.extend(response.json())

        # Parse Link header for the 'next' URL
        link_header = response.headers.get("Link", "")
        url = None
        for part in link_header.split(","):
            if 'rel="next"' in part:
                url = part.split(";")[0].strip().strip("<>")
                break

    return all_items

# Example: Fetch repos from GitHub
repos = fetch_all_link_header(
    "https://api.github.com/users/octocat/repos?per_page=5"
)
print(f"Total repos: {len(repos)}")
```

### Fetching training data from an API

```python
def fetch_training_data(api_url: str, num_samples: int = 1000) -> list:
    """Fetch labeled data from an API for training."""
    session = create_retry_session()
    data = []
    page = 1
    per_page = 100

    while len(data) < num_samples:
        response = session.get(
            api_url,
            params={"page": page, "per_page": per_page},
            timeout=15
        )
        response.raise_for_status()
        batch = response.json().get("data", [])

        if not batch:
            break

        data.extend(batch)
        page += 1

    session.close()
    return data[:num_samples]
```

### Rate limiting helper

```python
import time

class RateLimiter:
    """Simple rate limiter for API calls."""

    def __init__(self, calls_per_second: float = 5.0):
        self.min_interval = 1.0 / calls_per_second
        self.last_call = 0.0

    def wait(self):
        """Block until enough time has passed since the last call."""
        elapsed = time.time() - self.last_call
        if elapsed < self.min_interval:
            time.sleep(self.min_interval - elapsed)
        self.last_call = time.time()

# Usage
limiter = RateLimiter(calls_per_second=2)

for i in range(5):
    limiter.wait()
    response = requests.get(f"https://jsonplaceholder.typicode.com/posts/{i+1}")
    print(f"Request {i+1}: {response.status_code}")
```

## Common Mistakes

| Mistake | Fix |
|---------|-----|
| Hardcoding API keys in source code | Use `os.environ.get("KEY")` or a `.env` file |
| Forgetting `timeout=` | Always set a timeout; default is infinite wait |
| Not checking `status_code` | Call `response.raise_for_status()` after every request |
| Using `response.text` for JSON | Use `response.json()` which handles decoding |
| Ignoring rate limits (429 errors) | Implement exponential backoff or a rate limiter |
| Not closing sessions | Use `session.close()` or a context manager |
