# Day 41 - Testing with pytest

**Module:** Visualization, APIs & Testing | **Time:** ~60 min (35 read + 25 lab)

> "Code without tests is broken by design." In AI engineering, untested code leads to
> silent data corruption, wrong predictions, and models that pass evaluation but fail
> in production. pytest is the standard Python testing framework -- concise, powerful,
> and used by nearly every serious Python project including scikit-learn, pandas, and
> PyTorch.

## By the end of this lesson you will

- Write test functions, use assertions, and run them with pytest
- Use fixtures for setup/teardown and `@pytest.mark.parametrize` for multi-input testing
- Mock external dependencies (APIs, databases, file systems) with `unittest.mock` and `monkeypatch`

### Your first test

Create a file called `test_math_utils.py`:

```python
# math_utils.py
def add(a: float, b: float) -> float:
    return a + b

def divide(a: float, b: float) -> float:
    if b == 0:
        raise ValueError("Cannot divide by zero")
    return a / b

def sigmoid(x: float) -> float:
    import math
    return 1.0 / (1.0 + math.exp(-x))
```

```python
# test_math_utils.py
from math_utils import add, divide, sigmoid
import pytest
import math

def test_add_positive():
    assert add(2, 3) == 5

def test_add_negative():
    assert add(-1, -1) == -2

def test_add_float():
    assert add(0.1, 0.2) == pytest.approx(0.3)  # floating point!

def test_divide_normal():
    assert divide(10, 2) == 5.0

def test_divide_by_zero():
    with pytest.raises(ValueError, match="Cannot divide by zero"):
        divide(10, 0)

def test_sigmoid_zero():
    assert sigmoid(0) == pytest.approx(0.5)

def test_sigmoid_large_positive():
    assert sigmoid(100) == pytest.approx(1.0, abs=1e-6)

def test_sigmoid_large_negative():
    assert sigmoid(-100) == pytest.approx(0.0, abs=1e-6)
```


## Section 2: Assertions

pytest uses plain `assert` statements -- no special `assertEqual`, `assertTrue`, etc.

```python
# Equality
assert result == expected

# Truthiness
assert is_valid

# Containment
assert "error" in message
assert item in collection

# Type checking
assert isinstance(result, dict)

# Approximate float comparison
assert 0.1 + 0.2 == pytest.approx(0.3)
assert result == pytest.approx(expected, rel=1e-3)  # relative tolerance
assert result == pytest.approx(expected, abs=1e-6)  # absolute tolerance

# Exception testing
with pytest.raises(TypeError):
    add("a", 1)

with pytest.raises(ValueError, match="zero"):
    divide(1, 0)
```

## Section 3: Fixtures

Fixtures provide setup/teardown logic. They replace `setUp`/`tearDown` from unittest
with a more flexible, composable approach.

```python
# conftest.py -- fixtures defined here are available to ALL test files in the directory
import pytest
import sqlite3
import tempfile
import os

@pytest.fixture
def sample_data():
    """Provide sample training data."""
    return {
        "features": [[1, 2], [3, 4], [5, 6], [7, 8]],
        "labels": [0, 1, 1, 0]
    }

@pytest.fixture
def temp_db():
    """Create a temporary SQLite database for testing."""
    fd, path = tempfile.mkstemp(suffix=".db")
    conn = sqlite3.connect(path)
    conn.execute("""
        CREATE TABLE experiments (
            id INTEGER PRIMARY KEY,
            name TEXT,
            accuracy REAL
        )
    """)
    conn.commit()
    yield conn  # test runs here
    # Teardown
    conn.close()
    os.close(fd)
    os.unlink(path)

@pytest.fixture
def populated_db(temp_db):
    """Database with sample data (builds on temp_db)."""
    temp_db.executemany(
        "INSERT INTO experiments (name, accuracy) VALUES (?, ?)",
        [("exp_a", 0.85), ("exp_b", 0.91), ("exp_c", 0.78)]
    )
    temp_db.commit()
    return temp_db
```

### Using fixtures in tests

```python
# test_pipeline.py

def test_data_shape(sample_data):
    assert len(sample_data["features"]) == len(sample_data["labels"])

def test_db_insert(temp_db):
    temp_db.execute(
        "INSERT INTO experiments (name, accuracy) VALUES (?, ?)",
        ("test_exp", 0.90)
    )
    temp_db.commit()
    cursor = temp_db.execute("SELECT COUNT(*) FROM experiments")
    assert cursor.fetchone()[0] == 1

def test_best_accuracy(populated_db):
    cursor = populated_db.execute(
        "SELECT MAX(accuracy) FROM experiments"
    )
    best = cursor.fetchone()[0]
    assert best == pytest.approx(0.91)
```

### Fixture scope

```python
@pytest.fixture(scope="module")    # created once per test file
def expensive_model():
    # Imagine loading a large model
    return {"model": "loaded", "params": 1_000_000}

@pytest.fixture(scope="session")   # created once for the entire test run
def global_config():
    return {"batch_size": 32, "lr": 0.001}
```

| Scope | Lifetime |
|-------|----------|
| `function` (default) | Created fresh for each test |
| `class` | Once per test class |
| `module` | Once per test file |
| `session` | Once for the entire test run |

## Section 4: Parametrize

Test the same function with many inputs without writing separate test functions.

```python
import pytest

def normalize(value: float, min_val: float, max_val: float) -> float:
    """Min-max normalize a value to [0, 1]."""
    if max_val == min_val:
        raise ValueError("min and max cannot be equal")
    return (value - min_val) / (max_val - min_val)

@pytest.mark.parametrize("value, min_val, max_val, expected", [
    (5, 0, 10, 0.5),
    (0, 0, 10, 0.0),
    (10, 0, 10, 1.0),
    (-5, -10, 0, 0.5),
    (50, 0, 100, 0.5),
])
def test_normalize(value, min_val, max_val, expected):
    assert normalize(value, min_val, max_val) == pytest.approx(expected)

@pytest.mark.parametrize("value, min_val, max_val", [
    (5, 5, 5),   # min == max
])
def test_normalize_equal_bounds(value, min_val, max_val):
    with pytest.raises(ValueError):
        normalize(value, min_val, max_val)
```

## Section 5: Mocking

When your code calls an external API or database, you do not want your tests to depend
on those services. Mocking replaces real objects with controlled fakes.

### unittest.mock.patch

```python
# api_client.py
import requests

def get_model_prediction(text: str) -> str:
    """Call an external ML API."""
    response = requests.post(
        "https://api.example.com/predict",
        json={"text": text},
        timeout=10
    )
    response.raise_for_status()
    return response.json()["label"]
```

```python
# test_api_client.py
from unittest.mock import patch, MagicMock
from api_client import get_model_prediction

def test_get_prediction_positive():
    # Create a mock response object
    mock_response = MagicMock()
    mock_response.status_code = 200
    mock_response.json.return_value = {"label": "positive", "confidence": 0.95}
    mock_response.raise_for_status.return_value = None

    with patch("api_client.requests.post", return_value=mock_response) as mock_post:
        result = get_model_prediction("Great product!")

        assert result == "positive"
        mock_post.assert_called_once()
        # Verify the payload
        call_kwargs = mock_post.call_args
        assert call_kwargs.kwargs["json"]["text"] == "Great product!"

def test_get_prediction_api_error():
    mock_response = MagicMock()
    mock_response.raise_for_status.side_effect = Exception("503 Service Unavailable")

    with patch("api_client.requests.post", return_value=mock_response):
        import pytest
        with pytest.raises(Exception, match="503"):
            get_model_prediction("test")
```

### monkeypatch (pytest's built-in)

`monkeypatch` is simpler for straightforward patching and automatically restores
the original at the end of the test.

```python
# test_with_monkeypatch.py
import os

def get_api_key() -> str:
    key = os.environ.get("ML_API_KEY")
    if key is None:
        raise EnvironmentError("ML_API_KEY not set")
    return key

def test_get_api_key(monkeypatch):
    monkeypatch.setenv("ML_API_KEY", "test-key-123")
    assert get_api_key() == "test-key-123"

def test_get_api_key_missing(monkeypatch):
    monkeypatch.delenv("ML_API_KEY", raising=False)
    with pytest.raises(EnvironmentError):
        get_api_key()
```

### Mocking file reads

```python
def load_config(path: str) -> dict:
    import json
    with open(path) as f:
        return json.load(f)

def test_load_config(tmp_path):

    config = load_config(str(config_file))
```

## Section 7: AI Engineering Testing Patterns

### Testing a data pipeline

```python
# pipeline.py
import pandas as pd

def clean_data(df: pd.DataFrame) -> pd.DataFrame:
    """Clean a DataFrame for model training."""
    df = df.dropna(subset=["target"])
    df = df.drop_duplicates()
    df["text"] = df["text"].str.lower().str.strip()
    return df.reset_index(drop=True)
```

```python
# test_pipeline.py
import pandas as pd
from pipeline import clean_data

def test_clean_drops_null_target():
    df = pd.DataFrame({"text": ["hello", "world"], "target": [1, None]})
    result = clean_data(df)
    assert len(result) == 1
    assert result["target"].notna().all()

def test_clean_removes_duplicates():
    df = pd.DataFrame({"text": ["hello", "hello"], "target": [1, 1]})
    result = clean_data(df)
    assert len(result) == 1

def test_clean_lowercases_text():
    df = pd.DataFrame({"text": ["  HELLO World  "], "target": [1]})
    result = clean_data(df)
    assert result["text"].iloc[0] == "hello world"

def test_clean_preserves_schema():
    df = pd.DataFrame({"text": ["a", "b"], "target": [0, 1]})
    result = clean_data(df)
    assert list(result.columns) == ["text", "target"]
```

### Testing model predictions

```python
@pytest.mark.parametrize("input_text, expected_label", [
    ("I love this!", "positive"),
    ("I hate this!", "negative"),
    ("The weather is nice", "positive"),
])
def test_model_predictions(model_fixture, input_text, expected_label):

```

### Testing with pytest markers

```python
import pytest

@pytest.mark.slow
def test_full_training_pipeline():
    """This test takes minutes -- skip in CI fast mode."""
    pass

@pytest.mark.integration
def test_database_roundtrip():
    """Requires a real database connection."""
    pass

# Run only fast tests:  pytest -m "not slow"
# Run integration tests: pytest -m integration
```

### Task: Test a Data Processing Module

Create two files:

**`data_processor.py`:**
```python
import json
from pathlib import Path

class DataProcessor:
    def __init__(self, config_path: str = None):
        self.config = self._load_config(config_path) if config_path else {}

    def _load_config(self, path: str) -> dict:
        with open(path) as f:
            return json.load(f)

    def normalize(self, values: list[float]) -> list[float]:
        if not values:
            raise ValueError("Cannot normalize empty list")
        min_v, max_v = min(values), max(values)
        if min_v == max_v:
            return [0.0] * len(values)
        return [(v - min_v) / (max_v - min_v) for v in values]

    def remove_outliers(self, values: list[float], threshold: float = 2.0) -> list[float]:
        import statistics
        mean = statistics.mean(values)
        std = statistics.stdev(values)
        return [v for v in values if abs(v - mean) <= threshold * std]

    def process(self, values: list[float]) -> list[float]:
        cleaned = self.remove_outliers(values)
        return self.normalize(cleaned)
```

**`test_data_processor.py`** -- write tests covering:

## Common Mistakes

| Mistake | Fix |
|---------|-----|
| Using `assertEqual` instead of `assert` | pytest uses plain `assert`; no need for unittest methods |
| Comparing floats with `==` | Use `pytest.approx()` for float comparison |
| Tests depending on each other | Each test must be independent; use fixtures for shared setup |
| Mocking at the wrong location | Patch where the name is *used*, not where it is defined |
| No teardown for temp files | Use `tmp_path` fixture or `yield` in custom fixtures |
| Testing implementation, not behavior | Test public interfaces and outputs, not internal state |

### Solution 1: normalize() tests with parametrize

```python
# test_data_processor.py
import pytest
from data_processor import DataProcessor


@pytest.fixture
def processor():
    return DataProcessor()


@pytest.mark.parametrize("values, expected", [
    ([0, 5, 10],        [0.0, 0.5, 1.0]),
    ([10, 10, 10],      [0.0, 0.0, 0.0]),    # all-same values
    ([42.0],            [0.0]),               # single value (min == max)
    ([-10, 0, 10],      [0.0, 0.5, 1.0]),
    ([1, 2, 3, 4, 5],  [0.0, 0.25, 0.5, 0.75, 1.0]),
])
def test_normalize_parametrized(processor, values, expected):
    result = processor.normalize(values)
    assert result == pytest.approx(expected, abs=1e-6)
```

### Solution 3: remove_outliers() tests

```python
def test_remove_outliers_removes_extremes(processor):
    # 100 is a clear outlier in a dataset centered around 1-5
    values = [1, 2, 3, 4, 5, 100]
    result = processor.remove_outliers(values, threshold=2.0)
    assert 100 not in result
    assert all(v in result for v in [1, 2, 3, 4, 5])

def test_remove_outliers_keeps_all_normal(processor):
    values = [10, 11, 12, 13, 14]
    result = processor.remove_outliers(values, threshold=2.0)
    assert len(result) == len(values)

def test_remove_outliers_negative_outlier(processor):
    values = [1, 2, 3, 4, 5, -100]
    result = processor.remove_outliers(values, threshold=2.0)
    assert -100 not in result
```

### Solution 4: process() end-to-end

```python
def test_process_end_to_end(processor):
    # Clear outlier 1000 should be removed; remaining [1,2,3,4,5] normalized to [0..1]
    values = [1, 2, 3, 4, 5, 1000]
    result = processor.process(values)
    assert len(result) == 5
    assert min(result) == pytest.approx(0.0, abs=1e-6)
    assert max(result) == pytest.approx(1.0, abs=1e-6)
    assert all(0.0 <= v <= 1.0 for v in result)
```
